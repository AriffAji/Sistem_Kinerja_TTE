function jadikanSearchable(selectId) {
    const sel = document.getElementById(selectId);
    if (!sel || sel.dataset.ss) return;
    sel.dataset.ss = '1';

    const wrap = document.createElement('div');
    wrap.className = 'ss-wrap';
    const inp = document.createElement('input');
    inp.type = 'text';
    inp.className = 'ss-input';
    inp.autocomplete = 'off';
    inp.placeholder = 'Ketik nama untuk mencari...';
    const list = document.createElement('div');
    list.className = 'ss-list';

    sel.parentNode.insertBefore(wrap, sel);
    wrap.appendChild(inp);
    wrap.appendChild(sel);
    document.body.appendChild(list);
    sel.style.display = 'none';

    const descValue = Object.getOwnPropertyDescriptor(HTMLSelectElement.prototype, 'value');
    let aktifIdx = -1;

    function syncInput() {
        const o = sel.options[sel.selectedIndex];
        inp.value = (o && o.value && !o.disabled) ? o.text.trim() : '';
    }

    // Kode lain yang mengubah sel.value ikut menyegarkan kotak ketik
    Object.defineProperty(sel, 'value', {
        get() { return descValue.get.call(sel); },
        set(v) { descValue.set.call(sel, v); syncInput(); }
    });

    function render(q) {
        const kata = q.toLowerCase().split(/\s+/).filter(Boolean);
        let html = '', ada = false;
        Array.from(sel.options).forEach(o => {
            const t = o.text.trim();
            if (o.disabled) {
                if (kata.length === 0) html += `<div class="ss-group">${esc(t)}</div>`;
                return;
            }
            if (o.value === '') return;
            if (kata.every(k => t.toLowerCase().includes(k))) {
                ada = true;
                html += `<div class="ss-item" data-v="${esc(o.value)}">${esc(t)}</div>`;
            }
        });
        list.innerHTML = ada ? html : '<div class="ss-kosong">Tidak ada hasil</div>';
        aktifIdx = -1;
    }

    function buka() {
        const r = inp.getBoundingClientRect();
        const bawah = window.innerHeight - r.bottom - 10;
        list.style.left = r.left + 'px';
        list.style.top = (r.bottom + 2) + 'px';
        list.style.width = r.width + 'px';
        list.style.maxHeight = Math.max(120, Math.min(260, bawah)) + 'px';
        list.style.display = 'block';
    }
    function tutup() { list.style.display = 'none'; }
    function pilih(v) { sel.value = v; sel.dispatchEvent(new Event('change')); tutup(); inp.blur(); }
    function sorot(idx) {
        const items = list.querySelectorAll('.ss-item');
        if (items.length === 0) return;
        aktifIdx = (idx + items.length) % items.length;
        items.forEach((el, i) => el.classList.toggle('aktif', i === aktifIdx));
        items[aktifIdx].scrollIntoView({ block: 'nearest' });
    }

    inp.addEventListener('focus', () => { inp.select(); render(''); buka(); });
    inp.addEventListener('input', () => { render(inp.value); buka(); });
    inp.addEventListener('keydown', e => {
        if (e.key === 'ArrowDown') { e.preventDefault(); sorot(aktifIdx + 1); }
        else if (e.key === 'ArrowUp') { e.preventDefault(); sorot(aktifIdx - 1); }
        else if (e.key === 'Enter') {
            e.preventDefault();
            const items = list.querySelectorAll('.ss-item');
            const target = aktifIdx >= 0 ? items[aktifIdx] : (items.length === 1 ? items[0] : null);
            if (target) pilih(target.dataset.v);
        } else if (e.key === 'Escape') { tutup(); inp.blur(); }
    });
    inp.addEventListener('blur', () => {
        tutup();
        if (inp.value.trim() === '' && sel.value !== '') sel.value = '';
        else syncInput();
    });
    list.addEventListener('mousedown', e => e.preventDefault());
    list.addEventListener('click', e => { const el = e.target.closest('.ss-item'); if (el) pilih(el.dataset.v); });
    window.addEventListener('scroll', e => { if (!list.contains(e.target)) tutup(); }, true);
    window.addEventListener('resize', tutup);

    // Jika isi <select> diganti oleh kode lain (fetch), kotak ketik ikut menyesuaikan
    new MutationObserver(() => { syncInput(); if (list.style.display === 'block') render(inp.value); })
        .observe(sel, { childList: true });
    syncInput();
}

function aksiTugasLapor(item, perihalArg) {
    let html = '';
    const stId     = item.extra.surat_tugas_id;
    const stStatus = item.extra.surat_tugas_status;

    if (item.extra.jenis_tugas === 'rapat') {
        const rp = item.extra.rapat || {};
        if (rp.rapat_status_saya === 'belum') {
            html += rp.rapat_absen_buka
                ? `<button class="btn-disposisi" onclick="bukaModalAbsenRapat(${item.ref_id}, ${perihalArg})">✅ Absensi</button>`

            if (!rp.rapat_absen_buka && rp.rapat_ket_absen) {

            html += `<span style="font-size:0.78rem; color:#27ae60; font-weight:700;">✅ Sudah absen</span>`;

        if (rp.rapat_notulen && (!rp.rapat_laporan_status || rp.rapat_laporan_status === 'draft')) {
            html += `<button class="btn-tugaskan" style="background:#16a085;" onclick="bukaModalUbahJadwal(${item.ref_id}, ${perihalArg})">🕐 Ubah Jadwal</button>`;
        }
        if (rp.rapat_notulen) {
            const stLap = rp.rapat_laporan_status;

                html += `<button class="btn-tugaskan" style="background:#8e44ad;" onclick="bukaModalNotulensi(${item.ref_id}, ${perihalArg})">${stLap === 'revisi' ? '✏️ Revisi Notulensi' : '📝 Notulensi'}</button>`;

            const catatanEl = (step.catatan && step.jenis_ttd !== 'Mengetahui')
                ? `<div style="font-size:0.67rem; color:#7f8c8d; font-style:italic; margin-top:3px; padding-top:3px; border-top:1px dashed #ddd;">💬 ${esc(step.catatan)}</div>` : '';
            return `<div style="margin-bottom:5px; padding:6px 8px; background:#f8f9fa; border-radius:6px; border-left:3px solid ${clrBorder};">
                <div style="font-size:0.78rem; font-weight:700; color:#2c3e50;">${esc(step.nama)}</div>
                <div style="font-size:0.7rem; color:#7f8c8d;">${esc(step.jabatan)}</div>
                <div style="font-size:0.68rem; color:${clrBorder}; font-weight:700; margin-top:2px;">${icoJenis}${tglStep ? ' · ' + esc(tglStep) : ''}</div>

            </div>`;
        }).join('');
    }

}

let grupTerbuka = new Set();

async function muatRiwayatSayaLengkap() {
    try {
        const res = await fetch(`/api/riwayat-kasus?nip=${user.nip}`);
        cacheRiwayatSaya = await res.json();
        renderTabelRiwayat();
    } catch (err) {
        console.error('Gagal muat riwayat saya:', err);
        document.getElementById('tabelRiwayatSaya').innerHTML =
            '<tr><td colspan="5" style="text-align:center; color:#e74c3c;">Gagal memuat data.</td></tr>';
    }
}

function toggleGrup(key) {
    if (grupTerbuka.has(key)) grupTerbuka.delete(key); else grupTerbuka.add(key);
    renderTabelRiwayat();
}

function renderTabelRiwayat() {
    const filter = document.getElementById('filterRiwayat').value;
    const tbody  = document.getElementById('tabelRiwayatSaya');
    const badgeJenis = { eksternal: 'badge-eksternal', internal: 'badge-internal', permohonan: 'badge-permohonan' };
    const statusKasus = {
        selesai : { warna: '#27ae60', label: '✅ SELESAI' },
        berjalan: { warna: '#e67e22', label: '⏳ BERJALAN' },
        lain    : { warna: '#95a5a6', label: 'LAINNYA' },
    };
    const ikonTahap = { selesai: '✓', berjalan: '⏳', belum: '○' };

    const flat = [];      // dipakai tombol Tracking (indeks = urutan baris langkah yang tampil)
    let html = '';

    cacheRiwayatSaya.forEach(g => {
        const langkah = filter === 'semua' ? g.langkah : g.langkah.filter(i => i.tipe_aksi === filter);
        if (langkah.length === 0) return;

        const key  = String(g.kasus_id ?? 'lain');
        const buka = grupTerbuka.has(key);
        const sk   = statusKasus[g.status_kasus] || statusKasus.lain;
        const tglTerakhir = g.tanggal_terakhir ? g.tanggal_terakhir.split(' ')[0] : '—';
        const tunggu = g.menunggu || [];
        const teksTunggu = h => `${h.nama}: ${h.apa}${h.hari !== null ? ` (${h.hari} hari)` : ''}`;
        const stepper = (g.tahap || []).map(t => {
            const ket = tunggu.filter(h => h.tahap === t.kode).map(teksTunggu).join('\n');
            return `<span class="tahap tahap-${esc(t.status)}" title="${esc(ket)}">${ikonTahap[t.status] || ''} ${esc(t.label)}</span>`;
        }).join('<span style="color:#bdc3c7; font-size:0.7rem;">›</span>');
        const kotakTunggu = tunggu.length ? `
            <details class="menunggu-box" onclick="event.stopPropagation()" ${tunggu.length <= 2 ? 'open' : ''}>
                <summary>⏳ Sedang menunggu (${tunggu.length})</summary>
                ${tunggu.map(h => `<div class="menunggu-item">
                    <strong>${esc(h.nama)}</strong>${h.jabatan ? ` <span style="color:#95a5a6;">· ${esc(h.jabatan)}</span>` : ''}<br>
                    ${esc(h.apa)}${h.hari !== null ? ` <span class="hari">· ${h.hari} hari</span>` : ''}
                </div>`).join('')}
            </details>` : '';

        html += `<tr class="grup-baris ${buka ? 'terbuka' : ''}" onclick="toggleGrup(${arg(key)})">
            <td colspan="5">
                <div style="display:flex; justify-content:space-between; gap:12px; align-items:flex-start;">
                    <div style="min-width:0;">
                        <div class="grup-judul"><span class="caret">▶</span>
                            ${g.jenis_surat ? `<span class="badge ${badgeJenis[g.jenis_surat] || ''}" style="margin-right:8px;">${esc(g.jenis_surat.toUpperCase())}</span>` : ''}
                            ${esc(g.judul)}</div>
                        ${g.nomor_surat ? `<div class="grup-meta">${esc(g.nomor_surat)}</div>` : ''}
                        ${stepper ? `<div class="stepper">${stepper}</div>` : ''}
                        ${kotakTunggu}
                    </div>
                    <div style="text-align:right; white-space:nowrap;">
                        <span class="badge" style="background:${sk.warna};">${sk.label}</span>
                        ${g.kasus_id ? `<button class="btn-preview" style="background:#27ae60; margin:6px 0 0 8px;" onclick="event.stopPropagation(); bukaPohon(${g.kasus_id})">🌳 Lihat Alur</button>` : ''}
                        <div class="grup-meta">${langkah.length} langkah · terakhir ${esc(tglTerakhir)}</div>
                    </div>
                </div>
            </td></tr>`;

        if (!buka) return;

        langkah.forEach(item => {
            const idx = flat.push(item) - 1;
            const tgl = item.tanggal ? item.tanggal.split(' ')[0] : '—';

            let aksiHtml = '';
            if (item.surat_id) {
                aksiHtml += `<button class="btn-preview" onclick="previewSuratInbox(${item.surat_id})">👁️ Preview</button>
                             <button class="btn-preview" style="background:#8e44ad;" onclick="lihatTrackingRiwayat(${item.surat_id}, ${idx})">🔍 Tracking</button>`;
            }
            if (item.penerima_list) {
                item.penerima_list.filter(p => p.bisa_ditarik).forEach(p => {
                    const teks = item.penerima_list.length > 1
                        ? `↩️ Tarik: ${esc(p.nama.split(/[ ,]/)[0])}` : '↩️ Tarik Kembali';
                    aksiHtml += ` <button class="btn-preview" style="background:#c0392b;" onclick="tarikDisposisi(${p.disposisi_id})">${teks}</button>`;
                });
            }
            if (item.bisa_dikoreksi && item.penugasan_id) {
                aksiHtml += ` <button class="btn-preview" style="background:#e67e22;" onclick="bukaKoreksiTugas(${idx}, 'edit')">✏️ Edit Tugas</button>
                              <button class="btn-preview" style="background:#16a085;" onclick="bukaKoreksiTugas(${idx}, 'ganti')">🔁 Ganti Penerima</button>
                              <button class="btn-preview" style="background:#c0392b;" onclick="hapusPenugasan(${idx})">🗑️ Hapus</button>`;
            }
            if (item.url_notulen) {
                aksiHtml += ` <a href="${esc(item.url_notulen)}" target="_blank" class="btn-preview" style="background:#1a1a1a;">📄 Lihat Notulen</a>`;
            }
            if (item.file_daftar_hadir) {
                aksiHtml += ` <a href="/uploads/${esc(item.file_daftar_hadir)}" target="_blank" class="btn-preview" style="background:#1a1a1a;">📋 Lihat Daftar Hadir</a>`;
            }
            if (item.laporan_pd_id && item.file_laporan) {
                aksiHtml += ` <a href="/api/laporan-pd/${item.laporan_pd_id}/pdf" target="_blank" class="btn-preview" style="background:#27ae60;">📄 Laporan</a>`;
                if (!item.laporan_pd_final) {
                    aksiHtml += ` <button class="btn-preview" style="background:#e67e22;" onclick="bukaModalLaporPD(${item.penugasan_id}, ${arg(item.perihal)})">✏️ Edit</button>`;
                }
            }

            html += `<tr class="langkah-baris">
                <td style="font-size:0.82rem; white-space:nowrap;">${esc(tgl)}</td>
                <td>
                    <span title="${esc(item.label_aksi)}" style="background:#34495e; color:white; padding:3px 10px; border-radius:12px; font-size:0.72rem; font-weight:700; white-space:nowrap;">${esc(item.judul_ringkas || item.label_aksi)}</span>
                    ${item.detail_label ? `<div style="font-size:0.74rem; color:#7f8c8d; margin-top:5px;">${esc(item.detail_label)}</div>` : ''}
                    ${item.perihal && item.perihal !== g.judul ? `<div style="font-size:0.74rem; color:#95a5a6; margin-top:3px;">${esc(item.perihal)}</div>` : ''}
                </td>
                <td>${kolomPihakRiwayat(item)}</td>
                <td>${kolomCatatanRiwayat(item)}</td>
                <td style="vertical-align:top; padding:10px 14px;"><div class="aksi-wrap">${aksiHtml}</div></td>
            </tr>`;
        });
    });

    window._riwayatDataCache = flat;
    tbody.innerHTML = html || `<tr><td colspan="5"><div class="kosong-info"><span class="icon">📜</span><strong>Belum ada riwayat</strong></div></td></tr>`;
}


function kolomPihakRiwayat(item) {
    if (item.penerima_list && item.penerima_list.length > 0) {
        return item.penerima_list.map(p => `
            <div style="margin-bottom:6px;">
                <div style="font-size:0.85rem; font-weight:700; color:#2c3e50;">${esc(p.nama)}</div>
                <div style="font-size:0.75rem; color:#95a5a6;">${esc(p.jabatan || '')}</div>
            </div>`).join('');
    }
    return `<span style="font-size:0.85rem;">${esc(item.pihak_terkait)}</span><br><small style="color:#95a5a6;">${esc(item.jabatan_terkait || '')}</small>`;
}

function kolomCatatanRiwayat(item) {
    if (item.penerima_list && item.penerima_list.length > 1) {
        return item.penerima_list.map(p => `
            <div style="margin-bottom:6px;">
                <small style="color:#2c3e50; font-weight:700;">${esc(p.nama.split(',')[0])}:</small>
                <small style="color:#7f8c8d; font-style:italic;"> ${esc(p.catatan)}</small>
            </div>`).join('');
    }
    return `<small style="color:#7f8c8d; font-style:italic;">${esc(item.catatan || '—')}</small>`;
}


async function tarikDisposisi(disposisiId) {
    if (!confirm('Tarik disposisi ini? Disposisi akan hilang dari inbox penerima, lalu Anda bisa mengirim ulang ke orang yang benar.')) return;
    try {
        const res = await fetch(`/api/disposisi/${disposisiId}/tarik`, {
            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ nip_pengirim: user.nip })
        });
        const data = await res.json().catch(() => ({}));
        if (res.ok) { tampilNotif(data.message || 'Disposisi ditarik', 'sukses'); segarkanSemua(); }
        else tampilNotif(data.detail || 'Gagal menarik disposisi', 'gagal');
    } catch (e) { tampilNotif('Koneksi gagal!', 'gagal'); }
}

async function previewSuratInbox(suratId) {
    try {
        const res = await fetch(`/api/surat/${suratId}`);
        if (!res.ok) throw new Error('Gagal');
        const s = await res.json();
        const perihal = esc(s.perihal);

        // Surat upload PDF
        if (s.tipe_input === 'upload') {
            const urlPDF = s.file_final ? `/uploads/${s.file_final}` : (s.file_upload ? `/uploads/${s.file_upload}` : null);
            if (!urlPDF) { tampilNotif('File PDF tidak ditemukan!', 'gagal'); return; }
            window.open(urlPDF, '_blank');
            return;
        }

        // Surat eksternal: tampilkan file scan
        if (s.jenis_surat === 'eksternal') {
            if (!s.file_surat) { tampilNotif('File surat tidak ditemukan!', 'gagal'); return; }
            const ext = s.file_surat.split('.').pop().toLowerCase();
            const url = `/uploads/${s.file_surat}`;
            if (ext === 'pdf') { window.open(url, '_blank'); return; }
            const jendela = window.open('', '_blank', 'width=960,height=980');
            if (!jendela) { tampilNotif('Pop-up diblokir browser!', 'gagal'); return; }
            jendela.document.open();
            jendela.document.write(`<!DOCTYPE html>
<html><head><meta charset="UTF-8"><title>${perihal}</title>
<style>
    * { margin:0; padding:0; box-sizing:border-box; }
    body { background:#cbd5e0; }
    .toolbar { position:fixed; top:0; left:0; right:0; background:#2c3e50; color:white; padding:10px 24px; display:flex; align-items:center; justify-content:space-between; }
    .toolbar strong { display:block; font-size:0.95rem; }
    .toolbar span { font-size:0.78rem; color:rgba(255,255,255,0.65); }
    .btn-pdf { background:#2ecc71; color:white; border:none; padding:9px 24px; border-radius:6px; cursor:pointer; font-weight:bold; }
    .wrapper { padding:70px 20px 40px; display:flex; justify-content:center; }
    img { max-width:210mm; box-shadow:0 6px 30px rgba(0,0,0,0.25); display:block; }
</style></head><body>
    <div class="toolbar">
        <div><strong>${perihal}</strong><span>EKSTERNAL — ${esc(s.asal_surat || '—')}</span></div>
        <button class="btn-pdf" onclick="window.print()">🖨️ Cetak / PDF</button>
    </div>
    <div class="wrapper"><img src="${esc(url)}" alt="${perihal}"></div>
</body></html>`);
            jendela.document.close();
            return;
        }

        // Surat internal / permohonan (editor): render HTML A4
        let urlKop = null;
        const kodeKop = s.kode_unit_kop || s.kode_unit_pembuat;
        if (kodeKop) {
            const dataKop = await ambilJson(`/api/kop?kode_unit=${kodeKop}`, {});
            if (dataKop.ada) urlKop = dataKop.url;

    document.getElementById('modalTracking').classList.add('aktif');
    try {
        const res  = await fetch(`/api/surat/${suratId}/tracking`);
        const data = await res.json();
        document.getElementById('trackingPerihal').textContent = data.perihal;
        const formatTgl = tgl => tgl ? new Date(tgl).toLocaleString('id-ID', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' }) : null;

        let items = `<li class="timeline-item">
            <div class="timeline-dot awal">📝</div>

                <div class="tl-judul">Surat Dibuat</div>

                <div class="tl-waktu">${esc(formatTgl(data.tanggal_dibuat) || '—')}</div>
            </div></li>`;

            };
            const ico = icoMap[c.status] || { ico: '❓', cls: 'menunggu' };
            items += `<li class="timeline-item">
                <div class="timeline-dot ${ico.cls}">${ico.ico}</div>
                <div class="timeline-konten">
                    <div class="tl-judul">${esc(c.jenis_ttd)} <span style="font-size:0.72rem; color:#95a5a6; font-weight:normal;">— Urutan ${c.urutan}</span></div>
                    <div class="tl-subjudul">${esc(c.nama_pejabat)} — ${esc(c.jabatan_pejabat)}</div>
                    <div class="tl-waktu">${c.tanggal_proses ? esc(formatTgl(c.tanggal_proses)) : '<span style="color:#e67e22;">Belum diproses</span>'}</div>
                    ${c.catatan ? `<div class="tl-catatan">💬 ${esc(c.catatan)}</div>` : ''}
                </div></li>`;
        });

        if (data.tujuan && data.status === 'selesai') {
            items += `<li class="timeline-item">
                <div class="timeline-dot tujuan">📬</div>
                <div class="timeline-konten">
                    <div class="tl-judul">Diterima Penerima</div>
                    <div class="tl-subjudul">${esc(data.tujuan.nama)} — ${esc(data.tujuan.jabatan)}</div>

                </div></li>`;
        }

        // Node tambahan dari Riwayat Saya (disposisi, tugas, penilaian)
        if (extraCatatan && extraCatatan !== '—' && extraCatatan !== 'null') {
            const meta = {
                disposisi      : { ico: '📤', warna: '#2980b9', judul: 'Tindak Lanjut' },
                tugas          : { ico: '📝', warna: '#e67e22', judul: 'Penugasan' },

function bukaTTD(persetujuanId, jenisTTD, perihal, jenisSurat) {
    persetujuanAktif = { id: persetujuanId, jenis: jenisTTD, perihal, jenisSurat };
    const isKetahui = jenisTTD === 'Mengetahui';
    const isParaf   = jenisTTD === 'Paraf' || isKetahui;
    const btnKon = document.getElementById('btnKonfirmasiTTD');

    document.getElementById('modalTTDHeader').className = isParaf ? 'modal-header paraf' : 'modal-header';
    document.getElementById('modalTTDJudul').textContent    = isParaf ? '✍️ Konfirmasi Paraf' : '🔏 Konfirmasi TTE';
    document.getElementById('modalTTDSubjudul').textContent = isParaf
        ? 'Masukkan PIN untuk membubuhkan paraf' : 'Masukkan PIN untuk membubuhkan Tanda Tangan Elektronik';
    btnKon.className   = isParaf ? 'btn-konfirmasi paraf' : 'btn-konfirmasi';
    btnKon.textContent = isParaf ? '✍️ Paraf Sekarang' : '🔏 Tandatangani Sekarang';
    document.getElementById('modalTTDPerihal').textContent = perihal;

    // "Mengetahui" tidak memakai PIN
    document.getElementById('wrapPinTTD').style.display = isKetahui ? 'none' : 'block';
    if (isKetahui) {
        document.getElementById('modalTTDJudul').textContent    = '👁️ Konfirmasi Mengetahui';

        btnKon.textContent = '👁️ Saya Mengetahui';
    }
    document.getElementById('modalCatatan').value = '';

    // Penandatangan akhir surat permohonan wajib mengisi nomor surat
    const butuhNomor = !isParaf && jenisSurat === 'permohonan';
    document.getElementById('wrapNomorSuratInbox').style.display = butuhNomor ? 'block' : 'none';
    document.getElementById('inputNomorSuratInbox').value = '';

        el.value = '';
        el.className = isParaf ? 'pin-digit paraf' : 'pin-digit';

async function konfirmasiTTD() {
    if (!persetujuanAktif) return;
    const aktif = persetujuanAktif;
    const pin = ['m1','m2','m3','m4','m5','m6'].map(id => document.getElementById(id).value).join('');
    if (aktif.jenis !== 'Mengetahui' && pin.length !== 6) { tampilNotif('Masukkan PIN 6 digit terlebih dahulu!', 'gagal'); return; }

    const inputNomor = document.getElementById('inputNomorSuratInbox');
    const nomorSurat = inputNomor.value.trim();
    if (aktif.jenisSurat === 'permohonan' && aktif.jenis === 'TTE Utama' && !nomorSurat) {
        tampilNotif('Nomor surat wajib diisi sebelum TTE!', 'gagal');
        inputNomor.focus();
        return;
    }

    const btn = document.getElementById('btnKonfirmasiTTD');
    const teksAwal = btn.textContent;
    btn.textContent = '⏳ Memproses...';
    btn.disabled = true;

    const formData = new FormData();
    formData.append('nip', user.nip);
    formData.append('pin', aktif.jenis === 'Mengetahui' ? '-' : pin);
    formData.append('catatan', document.getElementById('modalCatatan').value);
    formData.append('nomor_surat', nomorSurat);
    try {
        const res = await fetch(`/api/tanda-tangan/${aktif.id}`, { method: 'POST', body: formData });
        const data = await res.json().catch(() => ({}));
        if (res.ok) {
            tutupModalTTD();
            tampilNotif(data.message, 'sukses');
            setTimeout(segarkanSemua, 800);
        } else {
            tampilNotif(data.detail || 'Gagal memproses!', 'gagal');
            ['m1','m2','m3','m4','m5','m6'].forEach(id => { document.getElementById(id).value = ''; });
            document.getElementById('m1').focus();
        }
    } catch (err) {
        tampilNotif('Koneksi gagal!', 'gagal');
    }
    btn.textContent = teksAwal;
    btn.disabled = false;
}

function tambahPenerimaTugas() {
    const select = document.getElementById('selectPenerimaDT');
    const nip  = select.value;
    const nama = select.options[select.selectedIndex]?.text || '—';
    if (!nip) { tampilNotif('Pilih penerima terlebih dahulu!', 'gagal'); return; }
    if (daftarPenugasanTerpilih.some(p => p.nip === nip)) { tampilNotif('Penerima ini sudah ada di daftar!', 'gagal'); return; }
    daftarPenugasanTerpilih.push({ nip, nama });
    select.value = '';
    renderDaftarPenerimaTugas();
}
function hapusPenerimaTugas(idx) { daftarPenugasanTerpilih.splice(idx, 1); renderDaftarPenerimaTugas(); }

        </div>`).join('');

async function kembalikanSurat(suratId) {
    const catatan = prompt('Alasan mengembalikan surat ke pembuat (wajib diisi):');
    if (catatan === null) return;
    if (!catatan.trim()) { tampilNotif('Alasan pengembalian wajib diisi!', 'gagal'); return; }
    try {
        const res = await fetch(`/api/surat/${suratId}/kembalikan`, {
            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ nip: user.nip, catatan: catatan.trim() })
        });
        const data = await res.json().catch(() => ({}));
        if (res.ok) { tampilNotif(data.message || 'Surat dikembalikan', 'sukses'); segarkanSemua(); }
        else tampilNotif(data.detail || 'Gagal mengembalikan surat', 'gagal');
    } catch (e) { tampilNotif('Koneksi gagal!', 'gagal'); }
}

    document.getElementById('wrapCatatanDisposisi').style.display = tipe === 'disposisi' ? 'block' : 'none';
    document.getElementById('wrapFormPenugasan').style.display    = tipe === 'penugasan' ? 'block' : 'none';
    if (tipe === 'disposisi') {
        document.getElementById('judulModalDT').textContent = 'Teruskan Surat (Disposisi)';
        btnKonfirmasi.style.background = '#2980b9';
        btnKonfirmasi.innerHTML = '📤 Kirim Disposisi';
    } else {
        document.getElementById('judulModalDT').textContent = 'Beri Tugas Langsung';
        btnKonfirmasi.style.background = '#e67e22';
        btnKonfirmasi.innerHTML = '📝 Beri Tugas';
    }

}

async function muatDropdownPenerimaDT(mode) {
    const select = document.getElementById('selectPenerimaDT');
    select.innerHTML = '<option value="">-- Memuat... --</option>';
    try {
        let url = `/api/kandidat-penerima?kode_unit=${user.kode_unit}&mode=${mode}`;
        if (mode === 'penugasan') url += `&nip_pemberi=${user.nip}`;
        const data = await (await fetch(url)).json();
        if (data.length === 0) { select.innerHTML = '<option value="">⚠️ Tidak ada kandidat penerima</option>'; return; }
        let opsi = '<option value="">-- Pilih Penerima --</option>';
        data.forEach(grup => {
            opsi += `<option disabled>── ${esc(grup.label)} ──</option>`;
            grup.pejabat.forEach(p => { opsi += `<option value="${esc(p.nip)}">${esc(p.nama)} — ${esc(p.nama_unit)}</option>`; });
        });
        select.innerHTML = opsi;
    } catch (e) {
        select.innerHTML = '<option value="">⚠️ Gagal memuat data</option>';
    }
}

async function konfirmasiDisposisiTugas() {
    const disposisi = dtTipeAksi === 'disposisi';
    if ((disposisi ? daftarPenerimaTerpilih : daftarPenugasanTerpilih).length === 0) {
        tampilNotif('Tambahkan minimal 1 penerima terlebih dahulu!', 'gagal');
        return;
    }
    const btn = document.getElementById('btnKonfirmasiDT');
    await kirimAksi(btn, '⏳ Mengirim...', async () => {
        const url  = disposisi ? '/api/disposisi' : '/api/penugasan';
        const body = disposisi ? {
            surat_id: dtSuratId, nip_pengirim: user.nip, disposisi_asal_id: dtDisposisiId,
            daftar_penerima: daftarPenerimaTerpilih.map(p => ({ nip_ke: p.nip, catatan: p.catatan }))
        } : {
            surat_id: dtSuratId, disposisi_id: dtDisposisiId, nip_pemberi: user.nip,
            daftar_penerima: daftarPenugasanTerpilih.map(p => p.nip),
            deskripsi_tugas: document.getElementById('deskripsiTugasInput').value || null,
            deadline: document.getElementById('deadlineTugasInput').value || null,
        };
        const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
        const data = await res.json().catch(() => ({}));
        if (res.ok) {
            tampilNotif(data.message || 'Berhasil diproses!', 'sukses');
            tutupModalDT();
            segarkanSemua();
        } else tampilNotif(data.detail || 'Gagal memproses!', 'gagal');
    });
}

async function konfirmasiLaporTugas() {
    const isiLaporan = document.getElementById('isiLaporanInput').value.trim();
    const fileInput  = document.getElementById('fileBuktiInput');
    const adaFile    = fileInput.files.length > 0;
    if (!adaFile && !laporSuratTugasId) { tampilNotif('File bukti (PDF) wajib dilampirkan!', 'gagal'); return; }
    if (adaFile && !fileInput.files[0].name.toLowerCase().endsWith('.pdf')) { tampilNotif('File bukti harus berformat PDF!', 'gagal'); return; }

    await kirimAksi(document.getElementById('btnKonfirmasiLapor'), '⏳ Mengirim...', async () => {
        let namaFileBukti = null;
        if (adaFile) {
            const formData = new FormData();
            formData.append('file', fileInput.files[0]);
            const resUp = await fetch('/api/upload/bukti-tugas', { method: 'POST', body: formData });
            if (!resUp.ok) throw new Error((await resUp.json().catch(() => ({}))).detail || 'Gagal upload file bukti');
            namaFileBukti = (await resUp.json()).nama_file;
        }
        const res = await fetch('/api/laporan-tugas', {
            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ penugasan_id: laporTugasPenugasanId, isi_laporan: isiLaporan || null, file_bukti: namaFileBukti })
        });
        if (res.ok) { tampilNotif('Laporan berhasil dikirim!', 'sukses'); tutupModalLaporTugas(); segarkanSemua(); }
        else tampilNotif((await res.json().catch(() => ({}))).detail || 'Gagal mengirim laporan!', 'gagal');
    });
}

let laporPDPenugasanId = null, laporPDLaporanId = null, ttdLaporanId = null;
let lampiranPD = [];        // { path, nama, baru }
let laporPDSiap = false;    // true hanya jika data laporan berhasil dimuat

async function ambilDataPD(penugasanId) {
    const res = await fetch(`/api/laporan-pd/data?penugasan_id=${penugasanId}&nip=${user.nip}`);
    const d = await res.json();
    if (!res.ok) throw new Error(d.detail || 'Gagal memuat data');
    return d;
}

function renderLampiranPD() {
    const wrap = document.getElementById('laporPDLampiran');
    if (lampiranPD.length === 0) { wrap.innerHTML = '<div style="font-size:0.78rem; color:#bdc3c7;">Belum ada lampiran.</div>'; return; }
    wrap.innerHTML = lampiranPD.map((l, i) => `
        <div class="baris-item">
            <a href="/uploads/${esc(l.path)}" target="_blank" style="font-size:0.82rem; color:#2980b9; overflow:hidden; text-overflow:ellipsis; white-space:nowrap;">📎 ${esc(l.nama)}</a>
            <button type="button" class="btn-x" onclick="hapusLampiranPD(${i})">✕</button>
        </div>`).join('');
}

function batalkanFilePD(path) {
    fetch('/api/upload/batalkan', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ nama_file: path }) }).catch(() => {});
}

async function unggahLampiranPD(input) {
    for (const f of Array.from(input.files)) {
        if (lampiranPD.length >= 10) { tampilNotif('Maksimal 10 lampiran', 'gagal'); break; }
        const fd = new FormData();
        fd.append('file', f);
        try {
            const res = await fetch('/api/upload/lampiran-pd', { method: 'POST', body: fd });
            const d = await res.json().catch(() => ({}));
            if (!res.ok) throw new Error(d.detail || 'Gagal mengunggah');
            lampiranPD.push({ path: d.nama_file, nama: d.nama_asli, baru: true });
            renderLampiranPD();
        } catch (e) { tampilNotif(`${f.name}: ${e.message}`, 'gagal'); }
    }
    input.value = '';
}

function hapusLampiranPD(i) {
    if (lampiranPD[i].baru) batalkanFilePD(lampiranPD[i].path);   // belum tersimpan: hapus dari server
    lampiranPD.splice(i, 1);
    renderLampiranPD();
}

async function bukaModalLaporPD(penugasanId, perihal) {
    laporPDSiap = false;
    laporPDPenugasanId = penugasanId;
    laporPDLaporanId = null;
    lampiranPD = [];
    renderLampiranPD();
    document.getElementById('laporPDPerihal').textContent = perihal;
    document.getElementById('laporPDInfo').textContent = 'Memuat data surat tugas...';
    document.getElementById('laporPDPeserta').innerHTML = '';
    document.getElementById('laporPDTempat').value = '';
    document.getElementById('laporPDHasil').value = '';
    document.getElementById('modalLaporPD').classList.add('aktif');
    try {
        const d = await ambilDataPD(penugasanId);
        const edit = !!d.laporan;
        if (edit && d.laporan.status !== 'draft') throw new Error('Laporan sudah final dan terkunci');
        laporPDLaporanId = edit ? d.laporan.id : null;

        document.getElementById('laporPDJudul').textContent = edit ? '✏️ Edit Laporan Perjalanan Dinas' : '✈️ Susun Laporan Perjalanan Dinas';
        document.getElementById('laporPDInfo').innerHTML = `
            <div><strong>Dasar:</strong> Surat Tugas Direktur Nomor ${esc(d.nomor_surat)} tanggal ${esc(d.tanggal_surat)}</div>

            <div><strong>Kegiatan:</strong> ${esc(d.kegiatan)}</div>`;

        document.getElementById('laporPDHasil').value  = edit ? d.laporan.hasil_kegiatan : '';
        lampiranPD = (d.lampiran || []).map(l => ({ path: l.path, nama: l.nama, baru: false }));
        renderLampiranPD();
        document.getElementById('laporPDCatatan').textContent = edit
            ? '⚠️ Semua peserta boleh mengedit selama laporan belum final. Menyimpan perubahan mereset tanda tangan yang sudah masuk: semua peserta menandatangani ulang.'
            : 'Setelah disusun, Anda dan peserta lain menandatangani laporan ini dengan PIN TTE. Peserta lain juga boleh mengedit sampai semua menandatangani.';

        document.getElementById('laporPDPeserta').innerHTML = d.peserta.map(p => {
            const saya = p.nip === user.nip;
            const ttd  = p.sudah_ttd ? ' <span style="color:#27ae60;">✅ sudah TTE</span>' : '';
            const aksi = saya
                ? '<span style="color:#95a5a6; font-size:0.75rem;">(Anda)</span>'
                : `<label style="font-size:0.75rem; color:#c0392b; cursor:pointer;"><input type="checkbox" class="cbKeluarkanPD" value="${esc(p.nip)}"> Tidak berangkat / keluarkan</label>`;

        }).join('');
        laporPDSiap = true;
    } catch (e) {
        document.getElementById('laporPDInfo').textContent = e.message || 'Gagal memuat data';
    }
}

function tutupModalLaporPD() {
    lampiranPD.filter(l => l.baru).forEach(l => batalkanFilePD(l.path));   // bersihkan unggahan yang tidak jadi disimpan
    lampiranPD = [];
    document.getElementById('modalLaporPD').classList.remove('aktif');
    laporPDPenugasanId = null;
    laporPDLaporanId = null;
    laporPDSiap = false;
}
pasangTutupLuar('modalLaporPD', tutupModalLaporPD);

async function konfirmasiLaporPD() {
    if (!laporPDSiap) { tampilNotif('Data laporan belum termuat. Tutup lalu buka lagi tombolnya.', 'gagal'); return; }
    const tempat = document.getElementById('laporPDTempat').value.trim();
    const hasil  = document.getElementById('laporPDHasil').value.trim();
    if (!tempat) { tampilNotif('Tempat wajib diisi!', 'gagal'); return; }
    if (!hasil)  { tampilNotif('Hasil kegiatan wajib diisi!', 'gagal'); return; }
    const keluar = Array.from(document.querySelectorAll('.cbKeluarkanPD:checked')).map(c => c.value);
    if (keluar.length > 0 && !confirm(`Keluarkan ${keluar.length} peserta dari laporan? Tugas mereka akan dibatalkan.`)) return;
    const lampiran = lampiranPD.map(l => ({ file: l.path, nama: l.nama }));

    await kirimAksi(document.getElementById('btnKonfirmasiLaporPD'), '⏳ Menyusun laporan...', async () => {
        const edit = laporPDLaporanId !== null;
        const res = await fetch(edit ? `/api/laporan-pd/${laporPDLaporanId}` : '/api/laporan-pd', {
            method: edit ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(edit
                ? { nip: user.nip, tempat, hasil_kegiatan: hasil, nip_dikeluarkan: keluar, lampiran }
                : { penugasan_id: laporPDPenugasanId, nip_pelapor: user.nip, tempat, hasil_kegiatan: hasil, nip_dikeluarkan: keluar, lampiran })
        });
        const data = await res.json().catch(() => ({}));
        if (res.ok) {
            tampilNotif(data.message || 'Laporan tersimpan', 'sukses');
            lampiranPD = [];       // sudah tersimpan: jangan dibersihkan saat modal ditutup
            tutupModalLaporPD();
            segarkanSemua();
        } else tampilNotif(data.detail || 'Gagal menyimpan laporan!', 'gagal');
    });
}

async function konfirmasiUbahJadwal() {
    const tanggal = document.getElementById('ubahJadwalTanggal').value;
    const mulai   = document.getElementById('ubahJadwalMulai').value;
    const selesai = document.getElementById('ubahJadwalSelesai').value;
    const alasan  = document.getElementById('ubahJadwalAlasan').value.trim();
    if (!tanggal) { tampilNotif('Tanggal rapat baru wajib diisi!', 'gagal'); return; }
    if (!mulai)   { tampilNotif('Jam mulai wajib diisi!', 'gagal'); return; }
    if (selesai && selesai <= mulai) { tampilNotif('Jam selesai harus setelah jam mulai!', 'gagal'); return; }
    if (!alasan)  { tampilNotif('Alasan perubahan jadwal wajib diisi!', 'gagal'); return; }

    await kirimAksi(document.getElementById('btnKonfirmasiUbahJadwal'), '⏳ Menyimpan...', async () => {
        const res = await fetch('/api/rapat/ubah-jadwal', {
            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                penugasan_id: ubahJadwalPenugasanId, nip: user.nip,
                tanggal, jam_mulai: mulai, jam_selesai: selesai || null, alasan
            })
        });
        const data = await res.json().catch(() => ({}));
        if (res.ok) { tampilNotif(data.message || 'Jadwal diperbarui', 'sukses'); tutupModalUbahJadwal(); segarkanSemua(); }
        else tampilNotif(data.detail || 'Gagal mengubah jadwal!', 'gagal');
    });
}

let notulenPenugasanId = null, notulenPeserta = [], notulenTamu = [];

function tautanPDFNotulen(urlNotulen, fh) {
    let h = '';
    if (urlNotulen) h += `<a href="${esc(urlNotulen)}" target="_blank" style="color:#8e44ad; font-weight:700; margin-right:16px;">📄 Lihat Notulen (PDF)</a>`;
    if (fh) h += `<a href="/uploads/${esc(fh)}" target="_blank" style="color:#27ae60; font-weight:700;">📋 Lihat Daftar Hadir (PDF)</a>`;
    document.getElementById('notulenLink').innerHTML = h || '<span style="color:#95a5a6;">PDF pratinjau muncul setelah Anda menekan Simpan Draft.</span>';
}

async function bukaModalNotulensi(penugasanId, perihal) {
    notulenPenugasanId = penugasanId;
    notulenPeserta = [];
    notulenTamu = [];
    document.getElementById('notulenPerihal').textContent = perihal;
    document.getElementById('notulenInfo').textContent = 'Memuat...';
    document.getElementById('notulenPeserta').innerHTML = '';
    document.getElementById('notulenTamu').innerHTML = '';
    document.getElementById('notulenHasil').value = '';
    document.getElementById('notulenRevisi').style.display = 'none';
    document.getElementById('modalNotulensi').classList.add('aktif');
    try {
        const res = await fetch(`/api/rapat/notulensi?penugasan_id=${penugasanId}&nip=${user.nip}`);
        const d = await res.json();
        if (!res.ok) throw new Error(d.detail || 'Gagal memuat data');

        document.getElementById('notulenInfo').innerHTML = `
            <div><strong>Agenda:</strong> ${esc(d.info.agenda)}</div>
            <div><strong>Hari/Tanggal:</strong> ${esc(d.info.hari_tanggal)}</div>
            <div><strong>Waktu:</strong> ${esc(d.info.waktu)}</div>
            <div><strong>Tempat:</strong> ${esc(d.info.tempat)}</div>
            <div><strong>Pimpinan Rapat:</strong> ${esc(d.info.pimpinan)}</div>`;
        if (d.catatan_revisi) {
            const rv = document.getElementById('notulenRevisi');
            rv.style.display = 'block';
            rv.innerHTML = `<strong>Catatan revisi pimpinan:</strong><br>${esc(d.catatan_revisi)}`;
        }
        notulenPeserta = d.peserta;
        notulenTamu = d.tamu;
        document.getElementById('notulenHasil').value = d.hasil_rapat || '';
        renderPesertaNotulen();
        renderTamuNotulen();
        tautanPDFNotulen(d.url_notulen, d.file_daftar_hadir);

        ['btnSimpanNotulen', 'btnKirimNotulen'].forEach(id => { document.getElementById(id).style.display = d.boleh_edit ? '' : 'none'; });
        document.getElementById('notulenHasil').disabled = !d.boleh_edit;
    } catch (e) {
        document.getElementById('notulenInfo').textContent = e.message || 'Gagal memuat data';
    }
}
function tutupModalNotulensi() {
    document.getElementById('modalNotulensi').classList.remove('aktif');
    notulenPenugasanId = null;
}
pasangTutupLuar('modalNotulensi', tutupModalNotulensi);

function renderPesertaNotulen() {
    document.getElementById('notulenPeserta').innerHTML = notulenPeserta.map((p, i) => {
        const nama = esc(p.nama) + (p.is_notulen ? ' <small style="color:#8e44ad;">(notulen)</small>' : '')
                   + (p.mewakili ? ` <small style="color:#7f8c8d;">(mewakili ${esc(p.mewakili)})</small>` : '');
        const kanan = p.editable
            ? `<select id="stNot_${i}" style="padding:5px; border:1px solid #ddd; border-radius:6px; font-size:0.78rem;">
                   <option value="belum" ${p.status === 'belum' ? 'selected' : ''}>Belum absen (otomatis Tidak hadir)</option>
                   <option value="izin" ${p.status === 'izin' ? 'selected' : ''}>Izin</option>
                   <option value="tidak_hadir" ${p.status === 'tidak_hadir' ? 'selected' : ''}>Tidak hadir</option>
               </select>
               <input type="text" id="alNot_${i}" value="${esc(p.alasan || '')}" placeholder="Alasan (opsional)" style="padding:5px; border:1px solid #ddd; border-radius:6px; font-size:0.78rem; width:150px;">`
            : `<span style="font-size:0.78rem; font-weight:700;">${esc(p.label)}</span>`;
        return `<div class="baris-item"><span style="font-size:0.84rem; font-weight:700; color:#2c3e50;">${nama}</span><span style="display:flex; gap:6px; align-items:center;">${kanan}</span></div>`;
    }).join('') || '<div style="color:#bdc3c7; font-size:0.82rem;">Belum ada peserta.</div>';
}

function renderTamuNotulen() {
    const wrap = document.getElementById('notulenTamu');
    if (notulenTamu.length === 0) { wrap.innerHTML = '<div style="color:#bdc3c7; font-size:0.82rem;">Belum ada peserta luar.</div>'; return; }
    wrap.innerHTML = notulenTamu.map((t, i) => `
        <div class="baris-item">
            <span style="font-size:0.84rem; font-weight:700; color:#2c3e50;">${esc(t.nama)}
                <small style="font-weight:normal; color:#7f8c8d;">${t.instansi ? '— ' + esc(t.instansi) : ''}</small></span>
            <span style="display:flex; gap:10px; align-items:center;">
                <label style="font-size:0.78rem; cursor:pointer;"><input type="checkbox" ${t.hadir ? 'checked' : ''} onchange="notulenTamu[${i}].hadir = this.checked"> Hadir</label>
                <button type="button" class="btn-x" onclick="hapusTamuNotulen(${i})">✕</button>
            </span>
        </div>`).join('');
}

function tambahTamuNotulen() {
    const nama = document.getElementById('tamuNama').value.trim();
    const ins  = document.getElementById('tamuInstansi').value.trim();
    if (!nama) { tampilNotif('Nama peserta luar wajib diisi!', 'gagal'); return; }
    notulenTamu.push({ nama, instansi: ins, hadir: true });
    document.getElementById('tamuNama').value = '';
    document.getElementById('tamuInstansi').value = '';
    renderTamuNotulen();
}
function hapusTamuNotulen(i) { notulenTamu.splice(i, 1); renderTamuNotulen(); }

function kumpulkanNotulensi() {
    const statusPeserta = [];
    notulenPeserta.forEach((p, i) => {
        if (!p.editable) return;
        statusPeserta.push({
            nip: p.nip,
            status_hadir: document.getElementById('stNot_' + i).value,
            alasan: document.getElementById('alNot_' + i).value.trim() || null
        });
    });
    return {
        penugasan_id: notulenPenugasanId, nip: user.nip,
        hasil_rapat: document.getElementById('notulenHasil').value,
        tamu: notulenTamu.map(t => ({ nama: t.nama, instansi: t.instansi || null, hadir: t.hadir })),
        status_peserta: statusPeserta
    };
}

async function simpanNotulensi(kirim) {
    const body = kumpulkanNotulensi();
    if (kirim) {
        if (!body.hasil_rapat.trim()) { tampilNotif('Hasil rapat wajib diisi!', 'gagal'); return; }
        if (!confirm('Kirim notulensi ke pimpinan rapat? Setelah dikirim tidak bisa diubah, dan peserta yang belum absen tercatat Tidak hadir.')) return;
    }
    await kirimAksi(document.getElementById(kirim ? 'btnKirimNotulen' : 'btnSimpanNotulen'), '⏳ Memproses...', async () => {
        const res = await fetch(kirim ? '/api/rapat/notulensi/kirim' : '/api/rapat/notulensi', {
            method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body)
        });
        const data = await res.json().catch(() => ({}));
        if (res.ok) {
            tampilNotif(data.message || 'Tersimpan', 'sukses');
            if (kirim) { tutupModalNotulensi(); segarkanSemua(); }
            else tautanPDFNotulen(data.url_notulen, data.file_daftar_hadir);
        } else tampilNotif(data.detail || 'Gagal memproses!', 'gagal');
    });
}

const IKON_POHON = { surat:'✉️', ttd:'🔏', disposisi:'📤', tugas:'📝', laporan:'📄', nilai:'⭐',
                     rapat:'📅', peserta:'👤', notulensi:'🗒️', laporan_pd:'✈️' };
const LABEL_STATUS_POHON = { selesai:'Selesai', berjalan:'Berjalan', belum:'Belum', batal:'Ditolak', info:'' };

function renderNodePohon(n, depth = 0) {
    const st   = n.status || 'info';
    const kls  = st === 'belum' ? 'info' : st;          // warna titik; "belum" tampil abu-abu
    const dot  = st === 'belum' ? '○' : (IKON_POHON[n.tipe] || '•');
    const aksi = [];
    if (n.surat_id) aksi.push(`<button class="btn-preview" onclick="event.stopPropagation(); previewSuratInbox(${n.surat_id})">👁️ Lihat Surat</button>`);
(n.files || []).forEach(f => aksi.push(`<a class="btn-preview" style="background:#1a1a1a;" href="${esc(f.url)}" target="_blank" onclick="event.stopPropagation()">📄 ${esc(f.label)}</a>`));

    const adaAnak = !!(n.anak && n.anak.length);
    const kepala = `<div class="pn-kepala">
        <span class="pn-caret">${adaAnak ? '▶' : ''}</span>
        <span class="pn-dot pn-${st === 'belum' ? 'x' : esc(st)}">${dot}</span>
        <div class="pn-isi">
            <div class="pn-judul">${esc(n.judul)}${LABEL_STATUS_POHON[st] ? `<span class="pn-badge pn-${esc(st)}">${LABEL_STATUS_POHON[st]}</span>` : ''}</div>
            ${n.sub ? `<div class="pn-sub">${esc(n.sub)}</div>` : ''}
            ${n.tanggal ? `<div class="pn-tgl">${esc(n.tanggal.replace('T', ' '))}</div>` : ''}
            ${aksi.length ? `<div class="pn-aksi">${aksi.join('')}</div>` : ''}
        </div></div>`;

    if (!n.anak || n.anak.length === 0) return `<div class="pn-daun">${kepala}</div>`;
    const buka = true;   // semua terbuka; pengguna bisa melipat sendiri
    return `<details class="pn" ${buka ? 'open' : ''}><summary>${kepala}</summary>
        <div class="pn-anak">${n.anak.map(a => renderNodePohon(a, depth + 1)).join('')}</div></details>`;
}

async function bukaPohon(suratId) {
    document.getElementById('pohonJudul').textContent = 'Memuat...';
    document.getElementById('pohonRingkas').innerHTML = '';
    document.getElementById('pohonIsi').innerHTML = '<div style="text-align:center; padding:24px; color:#95a5a6;">⏳ Menyusun alur...</div>';
    document.getElementById('modalPohon').classList.add('aktif');
    try {
        const res = await fetch(`/api/pohon-kasus/${suratId}?nip=${user.nip}`);
        const d = await res.json();
        if (!res.ok) throw new Error(d.detail || 'Gagal memuat alur');
        document.getElementById('pohonJudul').textContent = d.judul + (d.nomor_surat ? ` — ${d.nomor_surat}` : '');

        const ikon = { selesai: '✓', berjalan: '⏳', belum: '○' };
        const stepper = (d.tahap || []).map(t =>
            `<span class="tahap tahap-${esc(t.status)}">${ikon[t.status] || ''} ${esc(t.label)}</span>`
        ).join('<span style="color:#bdc3c7; font-size:0.7rem;"> › </span>');
        const tunggu = (d.menunggu || []).map(h => `<div class="menunggu-item">
            <strong>${esc(h.nama)}</strong>${h.jabatan ? ` <span style="color:#95a5a6;">· ${esc(h.jabatan)}</span>` : ''}<br>
            ${esc(h.apa)}${h.hari !== null ? ` <span class="hari">· ${h.hari} hari</span>` : ''}</div>`).join('');
        document.getElementById('pohonRingkas').innerHTML =
            `<div class="stepper">${stepper}</div>` +
            (tunggu ? `<div class="menunggu-box"><strong style="color:#b9770e;">⏳ Sedang menunggu (${d.menunggu.length})</strong>${tunggu}</div>` : '');
        document.getElementById('pohonIsi').innerHTML = renderNodePohon(d.pohon, 0);
    } catch (e) {
        document.getElementById('pohonIsi').innerHTML = `<div style="text-align:center; color:#e74c3c; padding:20px;">${esc(e.message || 'Gagal memuat alur')}</div>`;
    }
}

            const stepper = (k.tahap || []).map(t => `<span class="tahap tahap-${esc(t.status)}">${ikon[t.status] || ''} ${esc(t.label)}</span>`).join(' ');
            return `<tr>
                <td style="white-space:nowrap; font-size:0.82rem;">${esc((k.tanggal || '').split('T')[0].split(' ')[0])}</td>
                <td><span class="badge ${badgeJenis[k.jenis_surat] || ''}">${esc((k.jenis_surat || '').toUpperCase())}</span>
                    <strong style="margin-left:6px;">${esc(k.judul)}</strong><br>
                    <small style="color:#95a5a6;">${esc(k.nomor_surat || 'Belum ada nomor')} · dibuat ${esc(k.pembuat)}</small></td>
                <td><div class="stepper" style="margin:0;">${stepper}</div>
                    <span class="badge" style="background:${k.status_kasus === 'selesai' ? '#27ae60' : '#e67e22'}; margin-top:5px;">${k.status_kasus === 'selesai' ? '✅ SELESAI' : '⏳ BERJALAN'}</span></td>
                <td style="font-size:0.78rem; max-width:260px;">${k.menunggu_utama ? esc(k.menunggu_utama) + (k.jumlah_menunggu > 1 ? ` <small style="color:#95a5a6;">(+${k.jumlah_menunggu - 1} lainnya)</small>` : '') : '<span style="color:#95a5a6;">—</span>'}</td>
                <td><button class="btn-preview" style="background:#27ae60;" onclick="bukaPohon(${k.kasus_id})">🌳 Lihat Alur</button></td>
            </tr>`;
        }).join('');
    } catch (e) {
        tbody.innerHTML = `<tr><td colspan="5" style="text-align:center; color:#e74c3c;">${esc(e.message || 'Gagal memuat data.')}</td></tr>`;
    }
}

let koreksiAktif = null;   // { penugasanId, mode }

async function bukaKoreksiTugas(idx, mode) {
    const item = (window._riwayatDataCache || [])[idx];
    if (!item) return;
    koreksiAktif = { penugasanId: item.penugasan_id, mode };
    document.getElementById('koreksiPerihal').textContent = item.perihal;
    document.getElementById('koreksiWrapEdit').style.display  = mode === 'edit'  ? 'block' : 'none';
    document.getElementById('koreksiWrapGanti').style.display = mode === 'ganti' ? 'block' : 'none';
    document.getElementById('koreksiJudul').textContent = mode === 'edit' ? '✏️ Edit Tugas' : '🔁 Ganti Penerima';

    if (mode === 'edit') {
        document.getElementById('koreksiDeskripsi').value = item.deskripsi_tugas || '';
        document.getElementById('koreksiDeadline').value  = item.deadline ? String(item.deadline).slice(0, 10) : '';
    } else {
        document.getElementById('koreksiPenerimaLama').textContent = item.pihak_terkait || '—';
        const sel = document.getElementById('selectPenerimaKoreksi');
        sel.innerHTML = '<option value="">-- Memuat... --</option>';
        try {
            const data = await (await fetch(`/api/kandidat-penerima?kode_unit=${user.kode_unit}&mode=penugasan&nip_pemberi=${user.nip}`)).json();
            let opsi = '<option value="">-- Pilih Penerima --</option>';
            data.forEach(g => {
                opsi += `<option disabled>── ${esc(g.label)} ──</option>`;
                g.pejabat.forEach(p => { opsi += `<option value="${esc(p.nip)}">${esc(p.nama)} — ${esc(p.nama_unit)}</option>`; });
            });
            sel.innerHTML = opsi;
        } catch (e) {
            sel.innerHTML = '<option value="">⚠️ Gagal memuat data</option>';
        }
    }
    document.getElementById('modalKoreksiTugas').classList.add('aktif');
}

function tutupKoreksiTugas() {
    document.getElementById('modalKoreksiTugas').classList.remove('aktif');
    koreksiAktif = null;
}
pasangTutupLuar('modalKoreksiTugas', tutupKoreksiTugas);

async function konfirmasiKoreksiTugas() {
    if (!koreksiAktif) return;
    const { penugasanId, mode } = koreksiAktif;
    let url, metode, body;
    if (mode === 'edit') {
        const desk = document.getElementById('koreksiDeskripsi').value.trim();
        if (!desk) { tampilNotif('Deskripsi tugas tidak boleh kosong!', 'gagal'); return; }
        url = `/api/penugasan/${penugasanId}`; metode = 'PUT';
        body = { nip_pemberi: user.nip, deskripsi_tugas: desk,
                 deadline: document.getElementById('koreksiDeadline').value || null };
    } else {
        const nipBaru = document.getElementById('selectPenerimaKoreksi').value;
        if (!nipBaru) { tampilNotif('Pilih penerima baru terlebih dahulu!', 'gagal'); return; }
        url = `/api/penugasan/${penugasanId}/ganti-penerima`; metode = 'POST';
        body = { nip_pemberi: user.nip, nip_penerima_baru: nipBaru };
    }
    await kirimAksi(document.getElementById('btnKonfirmasiKoreksi'), '⏳ Menyimpan...', async () => {
        const res = await fetch(url, { method: metode, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
        const data = await res.json().catch(() => ({}));
        if (res.ok) { tampilNotif(data.message || 'Tersimpan', 'sukses'); tutupKoreksiTugas(); segarkanSemua(); }
        else tampilNotif(data.detail || 'Gagal menyimpan!', 'gagal');
    });
}

async function hapusPenugasan(idx) {
    const item = (window._riwayatDataCache || [])[idx];
    if (!item || !item.penugasan_id) return;
    if (!confirm(`Hapus penugasan untuk ${item.pihak_terkait}? Tugas ini akan hilang dari inbox penerima.`)) return;
    try {
        const res = await fetch(`/api/penugasan/${item.penugasan_id}/hapus`, {
            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ nip_pemberi: user.nip })
        });
        const data = await res.json().catch(() => ({}));
        if (res.ok) { tampilNotif(data.message || 'Penugasan dihapus', 'sukses'); segarkanSemua(); }
        else tampilNotif(data.detail || 'Gagal menghapus!', 'gagal');
    } catch (e) { tampilNotif('Koneksi gagal!', 'gagal'); }
}
