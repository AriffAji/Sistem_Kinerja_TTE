    function jadikanSearchable(selectId) {
        const sel = document.getElementById(selectId);
        if (!sel || sel.dataset.ss) return;
        sel.dataset.ss = '1';

        const wrap = document.createElement('div');
        wrap.className = 'ss-wrap';
        const inp = document.createElement('input');
        inp.type = 'text';
        inp.className = 'ss-input';
        inp.autocomplete = 'off';
        inp.placeholder = 'Ketik nama untuk mencari...';
        const list = document.createElement('div');
        list.className = 'ss-list';

        sel.parentNode.insertBefore(wrap, sel);
        wrap.appendChild(inp);
        wrap.appendChild(sel);
        document.body.appendChild(list);
        sel.style.display = 'none';

        const descValue = Object.getOwnPropertyDescriptor(HTMLSelectElement.prototype, 'value');
        let aktifIdx = -1;

        function syncInput() {
            const o = sel.options[sel.selectedIndex];
            inp.value = (o && o.value && !o.disabled) ? o.text.trim() : '';
        }

        // Kode lama yang mengubah sel.value (misal reset ke '') ikut menyegarkan kotak ketik
        Object.defineProperty(sel, 'value', {
            get() { return descValue.get.call(sel); },
            set(v) { descValue.set.call(sel, v); syncInput(); }
        });

        function esc(t) {
            return String(t).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
        }

        function render(q) {
            const kata = q.toLowerCase().split(/\s+/).filter(Boolean);
            let html = '';
            let ada = false;
            Array.from(sel.options).forEach(o => {
                const t = o.text.trim();
                if (o.disabled) {
                    if (kata.length === 0) html += `<div class="ss-group">${esc(t)}</div>`;
                    return;
                }
                if (o.value === '') return;
                const tl = t.toLowerCase();
                if (kata.every(k => tl.includes(k))) {
                    ada = true;
                    html += `<div class="ss-item" data-v="${esc(o.value)}">${esc(t)}</div>`;
                }
            });
            if (!ada) html = '<div class="ss-kosong">Tidak ada hasil</div>';
            list.innerHTML = html;
            aktifIdx = -1;
        }

        function buka() {
            const r = inp.getBoundingClientRect();
            const bawah = window.innerHeight - r.bottom - 10;
            list.style.left = r.left + 'px';
            list.style.top = (r.bottom + 2) + 'px';
            list.style.width = r.width + 'px';
            list.style.maxHeight = Math.max(120, Math.min(260, bawah)) + 'px';
            list.style.display = 'block';
        }

        function tutup() { list.style.display = 'none'; }

        function pilih(v) {
            sel.value = v;
            sel.dispatchEvent(new Event('change'));
            tutup();
            inp.blur();
        }

        function sorot(idx) {
            const items = list.querySelectorAll('.ss-item');
            if (items.length === 0) return;
            aktifIdx = (idx + items.length) % items.length;
            items.forEach((el, i) => el.classList.toggle('aktif', i === aktifIdx));
            items[aktifIdx].scrollIntoView({ block: 'nearest' });
        }

        inp.addEventListener('focus', () => { inp.select(); render(''); buka(); });
        inp.addEventListener('input', () => { render(inp.value); buka(); });
        inp.addEventListener('keydown', e => {
            if (e.key === 'ArrowDown') { e.preventDefault(); sorot(aktifIdx + 1); }
            else if (e.key === 'ArrowUp') { e.preventDefault(); sorot(aktifIdx - 1); }
            else if (e.key === 'Enter') {
                e.preventDefault();
                const items = list.querySelectorAll('.ss-item');
                const target = aktifIdx >= 0 ? items[aktifIdx] : (items.length === 1 ? items[0] : null);
                if (target) pilih(target.dataset.v);
            }
            else if (e.key === 'Escape') { tutup(); inp.blur(); }
        });
        inp.addEventListener('blur', () => {
            tutup();
            if (inp.value.trim() === '' && sel.value !== '') sel.value = '';
            else syncInput();
        });

        list.addEventListener('mousedown', e => e.preventDefault());   // jaga fokus agar klik terbaca
        list.addEventListener('click', e => {
            const el = e.target.closest('.ss-item');
            if (el) pilih(el.dataset.v);
        });

        window.addEventListener('scroll', e => { if (!list.contains(e.target)) tutup(); }, true);
        window.addEventListener('resize', tutup);

        // Jika isi <select> diganti oleh kode lain (fetch), kotak ketik ikut menyesuaikan
        new MutationObserver(() => {
            syncInput();
            if (list.style.display === 'block') render(inp.value);
        }).observe(sel, { childList: true });

        syncInput();
    }

        async function cekParameterLanjutan() {
            const params       = new URLSearchParams(window.location.search);
            const idAsal        = params.get('lanjutan_dari');
            const idDisposisi   = params.get('disposisi_dari');
            const sebagaiTugas  = params.get('sebagai') === 'tugas';
            const balasLangsung = params.get('balasan') === 'langsung';   // ✅ BARU
            if (!idAsal) return;

            try {
                const resAsal = await fetch(`/api/surat/${idAsal}`);
                if (!resAsal.ok) throw new Error('Surat asal tidak ditemukan');
                const asal = await resAsal.json();

                suratAsalId     = parseInt(idAsal);
                disposisiAsalId = idDisposisi ? parseInt(idDisposisi) : null;

                // Tampilkan banner
                document.getElementById('bannerSuratLanjutan').style.display = 'block';
                let labelBanner = asal.perihal;
                if (sebagaiTugas)  labelBanner = '📝 Surat Tugas untuk: ' + asal.perihal;
                if (balasLangsung) labelBanner = '↩️ Balasan untuk: ' + asal.perihal;
                document.getElementById('bannerSuratLanjutanPerihal').textContent = labelBanner;

                if (sebagaiTugas) {
                    // Surat Tugas: jenis = permohonan, TTE otomatis naik sampai Direktur
                    document.getElementById('pilih_jenis_surat').value = 'permohonan';
                    await gantiModeInput('permohonan');
                    document.getElementById('perihal').value = `Surat Tugas: ${asal.perihal}`;

                } else if (balasLangsung) {
                    // ✅ BARU — Balas langsung: internal, tujuan = pengirim surat asal (bukan Direktur)

                    document.getElementById('pilih_jenis_surat').value = 'internal';
                    await gantiModeInput('internal');
                    await muatUnitTujuan();

                    // ✅ FIX — pastikan pengirim asal selalu jadi opsi, walau dia tidak

                    // unit, atau ke-filter aturan level/punya_kop).
                    const selectTujuan = document.getElementById('nip_tujuan');
                    const sudahAda = Array.from(selectTujuan.options).some(o => o.value === asal.nip_pembuat);
                    if (!sudahAda) {
                        const resProfil = await fetch(`/api/profil?nip=${asal.nip_pembuat}`);
                        const namaTampil = resProfil.ok
                            ? (await resProfil.json()).nama
                            : asal.nama_pembuat || asal.nip_pembuat;

                        opsiBaru.value = asal.nip_pembuat;
                        opsiBaru.textContent = `${namaTampil} (Pengirim Surat Asal)`;
                        selectTujuan.appendChild(opsiBaru);

                    document.getElementById('perihal').value = `Balasan: ${asal.perihal}`;
                    tampilNotif(`Membalas surat dari: ${asal.perihal}`, 'sukses');

                } else {
                    // Surat lanjutan biasa (internal, ke ATASAN LANGSUNG)

        async function renderDelegasi() {
            const tbody = document.getElementById('tabelDelegasi');
            try {
                const res  = await fetch(`/api/delegasi?nip=${user.nip}`);
                if (!res.ok) throw new Error('Gagal');
                const data = await res.json();

                if (data.length === 0) {
                    tbody.innerHTML = '<tr><td colspan="4" style="text-align:center; color:#7f8c8d;">Tidak ada staf di unit Anda.</td></tr>';
                    return;
                }

                tbody.innerHTML = data.map(s => `
                    <tr>
                        <td><strong>${s.nama}</strong></td>
                        <td>${s.jabatan || '—'}</td>
                        <td>${s.aktif
                            ? '<span style="background:#27ae60; color:white; padding:3px 10px; border-radius:12px; font-size:0.72rem; font-weight:700;">✅ Ada Delegasi</span>'
                            : '<span style="background:#95a5a6; color:white; padding:3px 10px; border-radius:12px; font-size:0.72rem; font-weight:700;">Tanpa Delegasi</span>'}
                        </td>
                        <td>${s.aktif
                            ? `<button onclick="cabutDelegasi(${s.delegasi_id})"
                                   style="padding:5px 12px; background:#e74c3c; color:white; border:none; border-radius:4px; cursor:pointer;">
                                   ✕ Cabut
                               </button>`
                            : `<button onclick="beriDelegasi('${s.nip}')"
                                   style="padding:5px 12px; background:#2ecc71; color:white; border:none; border-radius:4px; cursor:pointer;">
                                   ➕ Beri Delegasi
                               </button>`}
                        </td>
                    </tr>`).join('');
                filterDelegasi();
            } catch (e) {
                tbody.innerHTML = '<tr><td colspan="4" style="text-align:center; color:#e74c3c;">Gagal memuat data.</td></tr>';
            }
        }

                function filterDelegasi() {
            const kata  = (document.getElementById('cariDelegasi')?.value || '').toLowerCase().split(/\s+/).filter(Boolean);
            const baris = document.querySelectorAll('#tabelDelegasi tr');
            let tampil = 0, total = 0;
            baris.forEach(tr => {
                if (tr.children.length < 4) return;          // lewati baris pesan (Memuat / kosong)
                total++;
                const teks = tr.textContent.toLowerCase();
                const cocok = kata.every(k => teks.includes(k));
                tr.style.display = cocok ? '' : 'none';
                if (cocok) tampil++;
            });
            const info = document.getElementById('infoJumlahDelegasi');
            if (info) info.textContent = total ? `Menampilkan ${tampil} dari ${total} staf` : '';

        async function muatDaftarPegawaiTugas() {
            const sel  = document.getElementById('pegawaiTugasSelect');
            const selP = document.getElementById('pimpinanRapatSelect');
            try {
                if (!cachePegawaiTugas) {
                    const res = await fetch('/api/pegawai-daftar');
                    if (!res.ok) throw new Error('Gagal');
                    cachePegawaiTugas = await res.json();
                }
                let opsi = '<option value="">-- Pilih Pegawai --</option>';
                cachePegawaiTugas.forEach(p => {
                    opsi += `<option value="${p.nip}">${p.nama} — ${p.jabatan || ''} (${p.nama_unit})</option>`;
                });
                sel.innerHTML = opsi;

                // Pimpinan rapat: semua pejabat non-staf
                let opsiP = '<option value="">-- Pilih Pimpinan Rapat --</option>';
                cachePegawaiTugas.filter(p => p.role_ppnp !== 'staf').forEach(p => {
                    opsiP += `<option value="${p.nip}">${p.nama} — ${p.jabatan || ''} (${p.nama_unit})</option>`;
                });
                selP.innerHTML = opsiP;
                selP.value = pimpinanRapatDipilih;
            } catch (e) {
                sel.innerHTML  = '<option value="">⚠️ Gagal memuat daftar pegawai</option>';
                selP.innerHTML = '<option value="">⚠️ Gagal memuat daftar pegawai</option>';
            }
            renderDaftarPegawaiTugas();
        }

        function tambahPegawaiTugas() {
            const sel  = document.getElementById('pegawaiTugasSelect');
            const nip  = sel.value;
            const nama = sel.options[sel.selectedIndex]?.text || '—';
            if (!nip) { tampilNotif('Pilih pegawai terlebih dahulu!', 'gagal'); return; }
            if (pegawaiTugasList.some(p => p.nip === nip)) {
                tampilNotif('Pegawai ini sudah ada di daftar!', 'gagal');
                return;
            }
            pegawaiTugasList.push({ nip, nama });
            sel.value = '';
            renderDaftarPegawaiTugas();
        }

        function hapusPegawaiTugas(idx) {
            pegawaiTugasList.splice(idx, 1);
            if (!pegawaiTugasList.some(p => p.nip === nipNotulen)) nipNotulen = null;
            renderDaftarPegawaiTugas();
        }

        function setNotulen(nip) {
            nipNotulen = nip;
        }

        function renderDaftarPegawaiTugas() {
            const wrap  = document.getElementById('daftarPegawaiTugas');
            const rapat = document.getElementById('jenisTugas').value === 'rapat';
            if (pegawaiTugasList.length === 0) {
                wrap.innerHTML = '<div style="text-align:center; padding:10px; color:#bdc3c7; font-size:0.82rem;">Belum ada pegawai ditambahkan.</div>';
                return;
            }
            wrap.innerHTML = pegawaiTugasList.map((p, i) => `
                <div style="background:white; border:1px solid #e0e6e4; border-radius:8px; padding:8px 12px;
                            margin-bottom:6px; display:flex; justify-content:space-between; align-items:center; gap:8px;">
                    <span style="font-size:0.85rem; font-weight:700; color:#2c3e50;">${i + 1}. ${p.nama}</span>
                    <span style="display:flex; align-items:center; gap:10px;">
                        ${rapat ? `<label style="font-size:0.78rem; color:#e67e22; font-weight:700; cursor:pointer;">
                            <input type="radio" name="radioNotulen" ${p.nip === nipNotulen ? 'checked' : ''}
                                   onchange="setNotulen('${p.nip}')"> Notulen
                        </label>` : ''}
                        <button type="button" onclick="hapusPegawaiTugas(${i})"
                            style="background:#fdf2f2; color:#e74c3c; border:1px solid #fcc; border-radius:4px;
                                   padding:3px 8px; cursor:pointer; font-size:0.75rem; font-weight:700;">✕</button>
                    </span>
                </div>`).join('');
        }

        function aturJenisTugas() {
            const jenis = document.getElementById('jenisTugas').value;
            const pd    = jenis === 'perjalanan_dinas';
            const rapat = jenis === 'rapat';

            if (!rapat) nipNotulen = null;
            renderDaftarPegawaiTugas();
        }

        async function muatDaftarSurat() {
            try {
                const res   = await fetch(`/api/surat?nip=${user.nip}&status=draft`);

                const tbody = document.getElementById('tabelDraft');

                if (data.length === 0) {
                    tbody.innerHTML =
                        '<tr><td colspan="5" style="text-align:center; color:#7f8c8d;">Belum ada draft surat.</td></tr>';
                    return;
                }

                const badgeWarna = { 'eksternal': '#2980b9', 'internal': '#8e44ad', 'permohonan': '#e67e22' };
                let html = '';

                data.forEach(s => {
                    const tgl   = s.tanggal_dibuat ? s.tanggal_dibuat.split('T')[0] : '—';
                    const nomor = s.nomor_surat    || null;
                    const kode  = s.kode_pelacakan || null;
                    const warna = badgeWarna[s.jenis_surat] || '#34495e';

                    let identHtml = '';
                    if (nomor && kode) {
                        identHtml = `<small style="color:#555;">${nomor}</small>
                                     <span style="color:#bdc3c7; margin:0 5px;">|</span>

                                     <span style="color:#bdc3c7; margin:0 5px;">|</span>
                                     <small style="font-family:monospace; font-weight:700; color:#27ae60;">${kode}</small>`;
                    } else if (nomor) {
                        identHtml = `<small style="color:#555;">${nomor}</small>`;
                    }

                    const isUpload = s.tipe_input === 'upload';
                    const kotakKembali = s.catatan_kembali
                        ? `<div style="margin-top:8px; background:#fdf2f2; border-left:4px solid #e74c3c;
                                       border-radius:4px; padding:8px 12px; font-size:0.8rem; color:#922b21; line-height:1.5;">

                    const aksiHtml = isUpload ? `
                        <button onclick="previewSurat(${s.id})"

                            👁️ Preview
                        </button>
                        <button onclick="bukaPosisiTTD(${s.id}, '${perihalEsc}', '${s.file_upload || ''}', '${s.jenis_surat}')"
                        style="padding:5px 10px; background:#8e44ad; color:white; border:none; border-radius:4px; cursor:pointer; margin-right:5px;">
                        📍 Atur Posisi TTD
                        </button>
                        <button onclick="konfirmasiHapus(${s.id})"
                            style="padding:5px 10px; background:#e74c3c; color:white; border:none; border-radius:4px; cursor:pointer; margin-right:5px;">
                            🗑️ Hapus
                        </button>
                        <button onclick="kirimSurat(${s.id})"
                            style="padding:5px 10px; background:#2ecc71; color:white; border:none; border-radius:4px; cursor:pointer;">
                            🚀 Kirim
                        </button>
                    ` : `
                        <button onclick="previewSurat(${s.id})"

                            🧾 Tanda Terima</button>` : ''}

        async function editSurat(id) {
            try {
                const res = await fetch(`/api/surat/${id}`);
                if (!res.ok) throw new Error('Gagal ambil data');
                const s = await res.json();

                suratEditId = id;

                document.getElementById('pilih_jenis_surat').value = s.jenis_surat;
                gantiModeInput(s.jenis_surat);

                document.getElementById('nomor_surat').value      = s.nomor_surat || '';
                document.getElementById('asal_surat').value       = s.asal_surat  || '';
                document.getElementById('perihal').value          = s.perihal     || '';
                document.getElementById('tanggal_surat').value    = s.tanggal_surat    || '';
                document.getElementById('tanggal_diterima').value = s.tanggal_diterima || '';

                if (s.jenis_surat === 'internal') {
                    // Tunggu dropdown tujuan selesai dimuat sebelum di-set nilainya
                    await muatUnitTujuan();
                    document.getElementById('nip_penandatangan').value = s.nip_penandatangan || '';
                    document.getElementById('nip_tujuan').value        = s.nip_tujuan        || '';
                    document.getElementById('tujuan_tambahan').value   = s.tujuan_tambahan   || '';
                }

                if (s.jenis_surat !== 'eksternal' && quillSurat) {
                    quillSurat.root.innerHTML = s.isi_surat || '';
                }

                if (s.jenis_surat === 'permohonan') {
                    const adaDataTugas = (s.pegawai_ditugaskan || []).length > 0 || s.tanggal_tugas_selesai;
                    if (adaDataTugas) {
                        document.getElementById('wrap_surat_tugas').style.display = 'block';
                        muatDaftarPegawaiTugas();
                    }
                    pegawaiTugasList = (s.pegawai_ditugaskan || []).map(p => ({ nip: p.nip, nama: p.nama }));
                    document.getElementById('tanggalTugasMulai').value   = s.tanggal_tugas_mulai   || '';
                    document.getElementById('tanggalTugasSelesai').value = s.tanggal_tugas_selesai || '';
                    document.getElementById('jenisTugas').value  = s.jenis_tugas  || '';
                    document.getElementById('tempatTugas').value = s.tempat_tugas || '';
                    document.getElementById('agendaRapat').value     = s.agenda_rapat || '';
                    document.getElementById('jamMulaiRapat').value   = s.jam_mulai    || '';
                    document.getElementById('jamSelesaiRapat').value = s.jam_selesai  || '';
                    pimpinanRapatDipilih = s.nip_pimpinan_rapat || '';
                    document.getElementById('pimpinanRapatSelect').value = pimpinanRapatDipilih;
                    const notulenLama = (s.pegawai_ditugaskan || []).find(p => p.is_notulen);
                    nipNotulen = notulenLama ? notulenLama.nip : null;
                    aturJenisTugas();
                    renderDaftarPegawaiTugas();
                }

                document.querySelector('.btn-submit').innerText = '💾 Simpan Perubahan';
                document.getElementById('formDetailSurat').scrollIntoView({ behavior: 'smooth' });

            } catch (err) {
                tampilNotif('Gagal memuat data draft untuk diedit!', 'gagal');
                console.error(err);
            }
        }

        async function eksekusiHapus() {
            if (!idSuratAkanDihapus) return;

            const btn      = document.getElementById('btnKonfirmasiHapus');
            btn.textContent = '⏳ Menghapus...';
            btn.disabled    = true;

            try {
                const res = await fetch(`/api/surat/${idSuratAkanDihapus}`, { method: 'DELETE' });
                if (res.ok) {
                    tutupModalHapus();
                    tampilNotif('Draft berhasil dihapus!', 'sukses');
                    muatDaftarSurat();
                } else {
                    const err = await res.json().catch(() => ({}));

                }
            } catch (err) {
                tampilNotif('Koneksi gagal saat menghapus!', 'gagal');
                console.error(err);
            }

            btn.textContent = '🗑️ Ya, Hapus';
            btn.disabled    = false;
        }

        async function simpanDraft(e) {
            e.preventDefault();
            // ✅ Cegah submit ganda kalau tombol diklik berkali-kali dengan cepat
    if (sedangMenyimpan) return;
    sedangMenyimpan = true;
            const btnSimpan     = document.querySelector('.btn-submit');
            btnSimpan.innerText = '⏳ Menyimpan...';
            btnSimpan.disabled  = true;

            let namaFileUploaded = null;

            // Tahap 1: Upload file (hanya eksternal)
            if (modeSurat === 'eksternal') {
                const fileInput = document.getElementById('file_surat');
                if (fileInput.files.length === 0 && !suratEditId) {
                    tampilNotif('File surat asli wajib diupload!', 'gagal');
                    btnSimpan.innerText = '💾 Simpan ke Draft (Preview)';
                    btnSimpan.disabled  = false;
                    sedangMenyimpan = false;
                    return;
                }

                const formData = new FormData();
                formData.append('file',        fileInput.files[0]);
                formData.append('nomor_surat', document.getElementById('nomor_surat').value || '');
                formData.append('asal_surat',  document.getElementById('asal_surat').value  || '');
                formData.append('perihal',     document.getElementById('perihal').value      || '');

                try {
                    const resUpload = await fetch('/api/upload/surat', { method: 'POST', body: formData });
                    if (resUpload.ok) {
                        namaFileUploaded = (await resUpload.json()).nama_file;
                    } else {
                        throw new Error('Gagal upload');
                    }
                } catch (err) {
                    tampilNotif('Gagal mengupload file!', 'gagal');
                    btnSimpan.innerText = '💾 Simpan ke Draft (Preview)';
                    btnSimpan.disabled  = false;
                    sedangMenyimpan = false;
                    return;
                }
            }

                        // Tahap 2: Upload PDF (hanya tipe upload)
            let namaFileUpload = null;
            if (tipeInputSurat === 'upload') {
                const fileInput = document.getElementById('inputFilePDF');
                if (fileInput.files.length === 0 && !suratEditId) {
                    tampilNotif('File PDF wajib diupload!', 'gagal');
                    btnSimpan.innerText = '💾 Simpan ke Draft (Preview)';
                    btnSimpan.disabled  = false;
                    sedangMenyimpan = false;
                    return;
                }

                if (fileInput.files.length > 0) {
    const formDataPDF = new FormData();
    formDataPDF.append('file', fileInput.files[0]);
    formDataPDF.append('jenis_surat', modeSurat);
    formDataPDF.append('perihal', document.getElementById('perihal').value || '');
    try {
        const resUpPDF = await fetch('/api/upload/dokumen', {
            method: 'POST', body: formDataPDF
        });
                        if (resUpPDF.ok) {
                            namaFileUpload = (await resUpPDF.json()).nama_file;
                        } else {
                            throw new Error('Gagal upload PDF');
                        }
                    } catch (err) {
                        tampilNotif('Gagal mengupload file PDF!', 'gagal');
                        btnSimpan.innerText = '💾 Simpan ke Draft (Preview)';
                        btnSimpan.disabled  = false;
                        sedangMenyimpan = false;
                        return;
                    }
                }
            }

            // Surat tugas dari penugasan: wajib ada pegawai yang ditugaskan
            if (modeSurat === 'permohonan' && penugasanAsalId && pegawaiTugasList.length === 0) {
                tampilNotif('Tambahkan minimal 1 pegawai yang ditugaskan (klik tombol ➕ Tambah)!', 'gagal');
                btnSimpan.innerText = '💾 Simpan ke Draft (Preview)';
                btnSimpan.disabled  = false;
                sedangMenyimpan     = false;
                return;
            }

            // Perjalanan dinas: wajib ada pegawai & tempat
            if (modeSurat === 'permohonan' && document.getElementById('jenisTugas').value === 'perjalanan_dinas') {
                const kurang = pegawaiTugasList.length === 0
                    ? 'Tambahkan minimal 1 pegawai yang ditugaskan (klik tombol ➕ Tambah)!'
                    : (!document.getElementById('tempatTugas').value.trim() ? 'Tempat pelaksanaan wajib diisi!' : '');
                if (kurang) {
                    tampilNotif(kurang, 'gagal');
                    btnSimpan.innerText = '💾 Simpan ke Draft (Preview)';
                    btnSimpan.disabled  = false;
                    sedangMenyimpan     = false;
                    return;
                }
            }

            // Rapat: wajib lengkap
            if (modeSurat === 'permohonan' && document.getElementById('jenisTugas').value === 'rapat') {
                const kurang =
                    pegawaiTugasList.length === 0 ? 'Tambahkan minimal 1 peserta rapat (klik tombol ➕ Tambah)!'

        function kirimSurat(id) {
            const baris   = document.querySelector(`button[onclick="kirimSurat(${id})"]`).closest('tr');
            const elPerihal = baris.querySelector('.teks-perihal');
            const perihal = elPerihal ? elPerihal.textContent : baris.querySelector('td:nth-child(3)').textContent;
            const jenis   = baris.querySelector('td:nth-child(2)').textContent.trim().toLowerCase();

            idSuratAkanDikirim = id;
            document.getElementById('kirimPerihal').textContent = perihal;

            // Teks berbeda berdasarkan jenis surat
            const teksKirim = document.getElementById('teksKirim');
            if (jenis.includes('eksternal')) {
                teksKirim.textContent = 'Surat berikut akan dikirim ke Direktur untuk diproses / Disposisi:';
            } else {
                teksKirim.textContent = 'Surat berikut akan dikirim untuk proses paraf / TTE:';

        }

            document.getElementById('modalKirim').classList.remove('aktif');

        }

        async function konfirmasiKirim() {
            if (!idSuratAkanDikirim) return;

            const btn      = document.getElementById('btnKonfirmasiKirim');
            btn.textContent = '⏳ Mengirim...';
            btn.disabled    = true;

            try {
                const res = await fetch(`/api/surat/${idSuratAkanDikirim}/kirim`, { method: 'POST' });

                if (res.ok) {
                    const data = await res.json();
                    tutupModalKirim();
                    tampilNotif("Surat berhasil dikirim ke pejabat!", "sukses");

                    if (data && data.chain_approval && data.chain_approval.length > 0) {
                        const infoChain = data.chain_approval
                            .map(c => `${c.urutan}. ${c.jenis}: ${c.nip}`)
                            .join('\n');
                        console.log("Chain approval:\n" + infoChain);
                    }

                    muatDaftarSurat();     // refresh tab draft (surat hilang dari sini)
                    muatSuratTerkirim();   // refresh tab terkirim
                } else {
                    const err = await res.json().catch(() => ({}));
                    tampilNotif(err.detail || "Gagal mengirim surat!", "gagal");
                }
            } catch (err) {
                tampilNotif("Koneksi gagal!", "gagal");
                console.error(err);
            }

            btn.textContent = '✅ Ya, Kirim Sekarang';
            btn.disabled    = false;
        }

        async function lihatTracking(suratId) {
            document.getElementById('trackingPerihal').textContent = 'Memuat...';
            document.getElementById('timelineTracking').innerHTML  =
                '<li style="text-align:center; padding:20px; color:#95a5a6;">⏳ Memuat riwayat...</li>';
            document.getElementById('modalTracking').classList.add('aktif');

            try {
                const res  = await fetch(`/api/surat/${suratId}/tracking`);
                const data = await res.json();

                document.getElementById('trackingPerihal').textContent = data.perihal;

                const formatTgl = (tgl) => {
                    if (!tgl) return null;
                    return new Date(tgl).toLocaleString('id-ID', {
                        day: 'numeric', month: 'short', year: 'numeric',
                        hour: '2-digit', minute: '2-digit'
                    });
                };

                let items = '';

                // Step 1: Dibuat
                items += `<li class="timeline-item">
                    <div class="timeline-dot awal">📝</div>
                    <div class="timeline-konten">
                        <div class="tl-judul">Surat Dibuat</div>
                        <div class="tl-subjudul">${data.nama_pembuat} — ${data.jabatan_pembuat}</div>
                        <div class="tl-waktu">${formatTgl(data.tanggal_dibuat) || '—'}</div>
                    </div>
                </li>`;

                // Step 2: Chain approval
                data.chain.forEach(c => {
                    const icoMap = {
                        'menunggu' : { ico: '⏳', cls: 'menunggu' },
                        'disetujui': { ico: c.jenis_ttd === 'Paraf' ? '✍️' : '🔏', cls: 'selesai' },
                        'ditolak'  : { ico: '❌', cls: 'ditolak' }
                    };
                    const ico = icoMap[c.status] || { ico: '❓', cls: 'menunggu' };

                    items += `<li class="timeline-item">
                        <div class="timeline-dot ${ico.cls}">${ico.ico}</div>
                        <div class="timeline-konten">
                            <div class="tl-judul">
                                ${c.jenis_ttd}
                                <span style="font-size:0.72rem; color:#95a5a6; font-weight:normal;">
                                    — Urutan ${c.urutan}
                                </span>
                            </div>
                            <div class="tl-subjudul">${c.nama_pejabat} — ${c.jabatan_pejabat}</div>
                            <div class="tl-waktu">
                                ${c.tanggal_proses
                                    ? formatTgl(c.tanggal_proses)
                                    : '<span style="color:#e67e22;">Belum diproses</span>'}
                            </div>
                            ${c.catatan ? `<div class="tl-catatan">💬 ${c.catatan}</div>` : ''}
                        </div>
                    </li>`;
                });

                // Step 3: Tujuan (jika internal dan sudah selesai)
                if (data.tujuan && data.status === 'selesai') {
                    items += `<li class="timeline-item">
                        <div class="timeline-dot tujuan">📬</div>
                        <div class="timeline-konten">
                            <div class="tl-judul">Diterima Penerima</div>
                            <div class="tl-subjudul">${data.tujuan.nama} — ${data.tujuan.jabatan}</div>
                            <div class="tl-waktu" style="color:#27ae60;">✅ Surat telah final & terkirim</div>
                        </div>
                    </li>`;
                }

                document.getElementById('timelineTracking').innerHTML = items;

            } catch (err) {
                document.getElementById('timelineTracking').innerHTML =
                    '<li style="text-align:center; color:#e74c3c; padding:20px;">Gagal memuat tracking.</li>';
            }
        }

        async function muatUnitTujuan() {
            try {
                const res   = await fetch(`/api/unit-tujuan?kode_unit=${user.kode_unit}`);
                const units = await res.json();

                if (units.length === 0) {
                    document.getElementById('nip_tujuan').innerHTML =

                    return;
                }

                const labelLevel = {
                    1: '── Pimpinan Institusi ──',
                    2: '── Wakil Direktur ──',
                    3: '── Jurusan / Bagian ──',
                    4: '── Sub Unit / Tim Kerja ──'
                };

                let opsi     = '<option value="">-- Pilih Unit / Pejabat Tujuan --</option>';
                let levelSkrg = null;

                units.forEach(u => {
                    if (u.level !== levelSkrg) {
                        levelSkrg = u.level;
                        const label = labelLevel[u.level] || `── Level ${u.level} ──`;
                        opsi += `<option disabled style="color:#95a5a6; font-style:italic;">
                                    ${label}

                    }
                    opsi += `<option value="${u.nip_pimpinan}">
                                ${u.nama_pimpinan} — ${u.nama_unit}
                             </option>`;
                });

                document.getElementById('nip_tujuan').innerHTML = opsi;
            } catch (e) {
                console.error("Gagal muat unit tujuan:", e);
                document.getElementById('nip_tujuan').innerHTML =
                    '<option value="">⚠️ Gagal memuat unit tujuan</option>';
            }

// HELPER: Ambil opsi HTML pejabat penandatangan (untuk surat internal)
// Level 4 -> tampilkan unit sendiri + atasan level 3
// Level 1/2/3 -> tampilkan unit sendiri saja
// ============================================================
async function ambilOpsiPejabatInternal() {
    let opsi = '';

    try {
        // Cek info level unit user
        const resInfo = await fetch(`/api/unit-info?kode_unit=${user.kode_unit}`);
        const infoUnit = await resInfo.json();

        // Pejabat unit sendiri
        const r1 = await fetch(`/api/pejabat?kode_unit=${user.kode_unit}`);
        const p1 = await r1.json();

        if (infoUnit.level === 4 && infoUnit.kode_induk) {
            // ── Level 4: tampilkan unit sendiri + atasan level 3 ──
            opsi += `<option disabled>── Unit Sendiri ──</option>`;
            if (p1.length === 0) {
                opsi += `<option disabled>Tidak ada pejabat di unit ini</option>`;
            } else {
                p1.forEach(p => {
                    opsi += `<option value="${p.nip}">${p.nama} — ${p.jabatan}</option>`;
                });
            }

            const r2 = await fetch(`/api/pejabat?kode_unit=${infoUnit.kode_induk}`);
            const p2 = await r2.json();
            if (p2.length > 0) {
                opsi += `<option disabled>── Atasan (${infoUnit.nama_induk || 'Level 3'}) ──</option>`;
                p2.forEach(p => {
                    opsi += `<option value="${p.nip}">${p.nama} — ${p.jabatan}</option>`;
                });
            }

        } else {
            // ── Level 1/2/3: unit sendiri saja ──
            if (p1.length === 0) {
                opsi += `<option value="">⚠️ Tidak ada pejabat di unit Anda</option>`;
            } else {
                p1.forEach(p => {
                    opsi += `<option value="${p.nip}">${p.nama} — ${p.jabatan}</option>`;
                });
            }
        }
    } catch (e) {
        console.error('Gagal ambil opsi pejabat internal:', e);
        opsi = `<option value="">⚠️ Gagal memuat daftar pejabat</option>`;
    }

    return opsi;
}

        async function isiDropdownPejabatTTD(jenisSurat) {
    try {
        let opsi = '<option value="">-- Pilih Pejabat --</option>';

        if (jenisSurat === 'permohonan') {
            // ── Permohonan: hanya Direktur. Paraf berjenjang di bawahnya
            //    dibuat otomatis oleh backend (lihat set_posisi_ttd). ──
            const r  = await fetch(`/api/pejabat?kode_unit=D001`);
            const p  = await r.json();
            const direkturList = p.filter(x => x.jabatan && x.jabatan.toLowerCase().includes('direktur')
                                              && !x.jabatan.toLowerCase().includes('wakil'));

            const daftar = direkturList.length > 0 ? direkturList : p;
            daftar.forEach(x => {
                opsi += `<option value="${x.nip}">${x.nama} — ${x.jabatan}</option>`;
            });

        } else {
            // ── Internal: unit sendiri, + atasan level 3 kalau level 4 ──
            opsi += await ambilOpsiPejabatInternal();
        }

        document.getElementById('formTTDPejabat').innerHTML = opsi;
    } catch (e) { console.error('Gagal isi dropdown TTD:', e); }
}

        // RENDER HALAMAN PDF
        // ============================================================
        async function renderHalamanPDF(nomHal) {
            if (!pdfJsDoc) return;
            nomHal       = Math.max(1, Math.min(nomHal, totalHalaman));
            halamanAktif = nomHal;

            const page     = await pdfJsDoc.getPage(nomHal);
            const canvas   = document.getElementById('pdfCanvas');
            const ctx      = canvas.getContext('2d');
            const viewport = page.getViewport({ scale: 1 });
            halamanPtWidth  = viewport.width;
            halamanPtHeight = viewport.height;
            const maxW     = Math.min(680, window.innerWidth * 0.55);
            const scale    = maxW / viewport.width;
            const vp       = page.getViewport({ scale });
            

            canvas.width  = vp.width;
            canvas.height = vp.height;

            await page.render({ canvasContext: ctx, viewport: vp }).promise;
            document.getElementById('halamanInfo').textContent = `Hal ${nomHal} / ${totalHalaman}`;
            renderMarkerDiCanvas();
            renderMarkerNomorDiCanvas();
            }

        // ============================================================
        // KLIK CANVAS → MUNCUL MINI FORM
        // ============================================================
        function klikPdfCanvas(e) {
    const canvas = document.getElementById('pdfCanvas');
    const rect   = canvas.getBoundingClientRect();
    const xPct   = parseFloat(((e.clientX - rect.left) / canvas.width  * 100).toFixed(2));
    const yPct   = parseFloat(((e.clientY - rect.top)  / canvas.height * 100).toFixed(2));

    if (jenisPinAktif === 'nomor') {
        pendingMarkerNomor = { hal: halamanAktif, x: xPct, y: yPct };
        document.getElementById('posisiKoordLabelNomor').textContent =
            `Hal ${halamanAktif} — X: ${xPct.toFixed(1)}%  Y: ${yPct.toFixed(1)}%`;
        document.getElementById('formNomorPrefix').value = '';
        document.getElementById('modalFormNomor').classList.add('aktif');
        return;
    }

    if (jenisSuratPosisi === 'permohonan' && posisiNomorList.length === 0) {
        tampilNotif('Pasang minimal 1 pin Nomor Surat terlebih dahulu!', 'gagal');
        setJenisPin('nomor');

function gerakPdfCanvas(e) {
    if (jenisPinAktif === 'nomor') {

    sembunyikanGhostNomor();

    const canvas = document.getElementById('pdfCanvas');

    const frameWpx = FRAME_PT_W * (canvas.width  / halamanPtWidth);
    const frameHpx = FRAME_PT_H * (canvas.height / halamanPtHeight);

    const leftPx = e.clientX - rect.left;

    ghost.style.height  = `${frameHpx}px`;

        function renderMarkerDiCanvas() {
    const container = document.getElementById('markerContainer');
    // Hapus hanya frame & dot yang sudah terpasang, JANGAN hapus ghost frame
    container.querySelectorAll('.marker-frame, .marker-dot').forEach(el => el.remove());

    const canvas = document.getElementById('pdfCanvas');

    posisiTTDList.forEach((m, i) => {
        if (m.posisi_hal !== halamanAktif) return;

        const frameWpx = FRAME_PT_W * (canvas.width  / halamanPtWidth);
        const frameHpx = FRAME_PT_H * (canvas.height / halamanPtHeight);

        const leftPx = (m.posisi_x / 100) * canvas.width;
        const topPx  = (m.posisi_y / 100) * canvas.height;

        const frame = document.createElement('div');
        frame.className   = 'marker-frame';
        frame.style.left  = `${leftPx}px`;
        frame.style.top   = `${topPx}px`;
        frame.style.width  = `${frameWpx}px`;
        frame.style.height = `${frameHpx}px`;
        container.appendChild(frame);

        const dot = document.createElement('div');
        dot.className = 'marker-dot';
        dot.style.left = `${m.posisi_x}%`;
        dot.style.top  = `${m.posisi_y}%`;
        dot.textContent = i + 1;
        container.appendChild(dot);
    });
}

        // SIMPAN POSISI TTD KE SERVER
        // ============================================================
async function simpanPosisiTTD() {
    if (!suratPosisiId) return;
    const isPermohonan = jenisSuratPosisi === 'permohonan';

    if (isPermohonan && posisiNomorList.length === 0) {
        tampilNotif('Pin Nomor Surat wajib dipasang terlebih dahulu!', 'gagal');
        setJenisPin('nomor');
        return;
    }
    if (posisiTTDList.length === 0) {
        tampilNotif('Tambahkan minimal 1 posisi TTD terlebih dahulu!', 'gagal');
        return;
    }

    try {
        // 1) Nomor surat lebih dulu (khusus permohonan)
        if (isPermohonan) {
            const resNomor = await fetch(`/api/surat/${suratPosisiId}/posisi-nomor`, {
                method : 'POST',
                headers: { 'Content-Type': 'application/json' },
                body   : JSON.stringify({
                    posisi_list: posisiNomorList.map(m => ({
                        posisi_hal : m.posisi_hal,
                        posisi_x   : m.posisi_x,
                        posisi_y   : m.posisi_y,
                        prefix_teks: m.prefix_teks,
                    }))
                })
            });
            if (!resNomor.ok) {
                const err = await resNomor.json().catch(() => ({}));
                tampilNotif(err.detail || 'Gagal menyimpan posisi nomor surat!', 'gagal');
                return;
            }
        }

        // 2) Baru posisi TTD
        const res = await fetch(`/api/surat/${suratPosisiId}/posisi-ttd`, {
            method : 'POST',
            headers: { 'Content-Type': 'application/json' },
            body   : JSON.stringify({
                posisi_list : posisiTTDList.map(m => ({
                    nip_penyetuju: m.nip_penyetuju,
                    label_ttd    : m.label_ttd,
                    posisi_hal   : m.posisi_hal,
                    posisi_x     : m.posisi_x,
                    posisi_y     : m.posisi_y,
                })),
                mode_paralel: modeTTD === 'paralel'
            })
        });
        if (!res.ok) {
            const err = await res.json().catch(() => ({}));
            tampilNotif(err.detail || 'Gagal menyimpan posisi TTD!', 'gagal');
            return;
        }

        tampilNotif('Posisi berhasil disimpan! Surat siap dikirim.', 'sukses');
        tutupModalPosisi();
        muatDaftarSurat();

    } catch (err) {
        tampilNotif('Koneksi gagal!', 'gagal');
        console.error(err);
    }
}

(function () {
    const sel     = document.getElementById('pilih_jenis_surat');
    const btn     = document.getElementById('jenisDropdownBtn');
    const menu    = document.getElementById('jenisDropdownMenu');
    const wrap    = document.getElementById('jenisDropdown');
    const label   = document.getElementById('jenisDropdownLabel');
    const ikon    = document.getElementById('jenisDropdownIkon');
    const chevron = document.getElementById('jenisDropdownChevron');
    const items   = menu.querySelectorAll('[data-value]');

    function syncLabel() {
        const aktif = [...items].find(i => i.dataset.value === sel.value);
        if (aktif) {
            label.textContent = aktif.dataset.nama;
            ikon.textContent  = aktif.dataset.ikon;
            label.classList.replace('text-gray-500', 'text-gray-900');
        } else {
            label.textContent = '-- Pilih Kategori Surat --';
            ikon.textContent  = 'category';
            label.classList.replace('text-gray-900', 'text-gray-500');
        }
    }

    // Kalau kode lain mengubah sel.value (reset, surat lanjutan), label ikut berubah
    const descValue = Object.getOwnPropertyDescriptor(HTMLSelectElement.prototype, 'value');
    Object.defineProperty(sel, 'value', {
        get() { return descValue.get.call(this); },
        set(v) { descValue.set.call(this, v); syncLabel(); }
    });

    function buka(on) {
        menu.classList.toggle('hidden', !on);
        chevron.classList.toggle('rotate-180', on);
        btn.setAttribute('aria-expanded', on);
    }

    btn.addEventListener('click', e => { e.stopPropagation(); buka(menu.classList.contains('hidden')); });

    items.forEach(it => it.addEventListener('click', () => {
        sel.value = it.dataset.value;
        sel.dispatchEvent(new Event('change'));   // memicu gantiModeInput()
        buka(false);
    }));

    document.addEventListener('click', e => { if (!wrap.contains(e.target)) buka(false); });
    document.addEventListener('keydown', e => { if (e.key === 'Escape') buka(false); });

    syncLabel();
})();
